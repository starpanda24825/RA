local function srv_safeYield()
    if type(task) == "table" and type(task.wait) == "function" then
        task.wait()
        return
    end
    if type(wait) == "function" then
        wait()
        return
    end
    if type(coroutine) == "table" and type(coroutine.running) == "function" and type(coroutine.yield) == "function" then
        local running = coroutine.running()
        if running then pcall(coroutine.yield, 0) end
    end
end
local function srv_rightRotate(n, b)
    return bit32.band(bit32.bor(bit32.rshift(n, b), bit32.lshift(n, 32 - b)), 0xFFFFFFFF)
end

local function srv_rightShift(n, b)
    return bit32.rshift(n, b)
end

local function srv_toBytes(str)
    local bytes = {}
    for i = 1, #str do bytes[#bytes + 1] = string.byte(str:sub(i, i)) end
    return bytes
end

local function srv_ASHA512(password, salt, iterations)
    salt = salt or ""
    iterations = iterations or 1
    local function run_once(message)
        local h = {0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19}
        local function compress(chunk)
            local w = {}
            for i = 1, 16 do
                local o = (i - 1) * 4 + 1
                w[i] = bit32.band(bit32.bor(bit32.lshift(chunk[o] or 0, 24), bit32.lshift(chunk[o+1] or 0, 16), bit32.lshift(chunk[o+2] or 0, 8), (chunk[o+3] or 0)), 0xFFFFFFFF)
            end
            for i = 17, 64 do
                local s0 = bit32.bxor(srv_rightRotate(w[i-15],7), srv_rightRotate(w[i-15],18), srv_rightShift(w[i-15],3))
                local s1 = bit32.bxor(srv_rightRotate(w[i-2],17), srv_rightRotate(w[i-2],19), srv_rightShift(w[i-2],10))
                w[i] = bit32.band((w[i-16] + s0 + w[i-7] + s1), 0xFFFFFFFF)
            end
            local a,b,c,d,e,f,g,hh = h[1],h[2],h[3],h[4],h[5],h[6],h[7],h[8]
            for i = 1, 64 do
                local S1 = bit32.bxor(srv_rightRotate(e,6), srv_rightRotate(e,11), srv_rightRotate(e,25))
                local ch = bit32.bxor(bit32.band(e,f), bit32.band(bit32.bnot(e), g))
                local temp1 = bit32.band((hh + S1 + ch + srv_k[i] + w[i]), 0xFFFFFFFF)
                local S0 = bit32.bxor(srv_rightRotate(a,2), srv_rightRotate(a,13), srv_rightRotate(a,22))
                local maj = bit32.bxor(bit32.band(a,b), bit32.band(a,c), bit32.band(b,c))
                local temp2 = bit32.band((S0 + maj), 0xFFFFFFFF)
                hh = g; g = f; f = e; e = bit32.band((d + temp1), 0xFFFFFFFF)
                d = c; c = b; b = a; a = bit32.band((temp1 + temp2), 0xFFFFFFFF)
            end
            h[1] = bit32.band(h[1] + a, 0xFFFFFFFF)
            h[2] = bit32.band(h[2] + b, 0xFFFFFFFF)
            h[3] = bit32.band(h[3] + c, 0xFFFFFFFF)
            h[4] = bit32.band(h[4] + d, 0xFFFFFFFF)
            h[5] = bit32.band(h[5] + e, 0xFFFFFFFF)
            h[6] = bit32.band(h[6] + f, 0xFFFFFFFF)
            h[7] = bit32.band(h[7] + g, 0xFFFFFFFF)
            h[8] = bit32.band(h[8] + hh, 0xFFFFFFFF)
            local hash = ""
            for i = 1, 8 do hash = hash .. string.format("%08x", h[i]) end
            return hash
        end
        local msgBytes = srv_toBytes(message)
        local originalBitLength = #message * 8
        msgBytes[#msgBytes + 1] = 0x80
        while (#msgBytes % 64) ~= 56 do msgBytes[#msgBytes + 1] = 0x00 end
        for i = 1, 4 do msgBytes[#msgBytes + 1] = 0 end
        for i = 3, 0, -1 do
            msgBytes[#msgBytes + 1] = bit32.band(bit32.rshift(originalBitLength, i * 8), 0xFF)
        end
        local chunk = {}
        for j = 1, 64 do chunk[j] = msgBytes[j] or 0 end
        return compress(chunk)
    end
    local current = password
    for i = 1, iterations do
        if i % 100 == 0 then srv_safeYield() end
        current = run_once(current .. salt)
    end
    return current
end

srv_hashPassword = function(password, customSalt, iterations)
    local salt = customSalt or ""
    if salt == "" then
        math.randomseed(os.time())
        local chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789+/"
        for i = 1, 32 do
            local idx = math.random(1, #chars)
            salt = salt .. chars:sub(idx, idx)
        end
    end
    local iters = iterations or 1
    local hash = srv_ASHA512(password, salt, iters)
    return hash .. ":" .. salt .. ":" .. tostring(iters)
end

srv_verify = function(password, fullHash)
    local storedHash, salt, iters = tostring(fullHash or ""):match("([^:]+):([^:]+):(%d+)")
    if not storedHash or not salt or not iters then return false end
    local computed = srv_ASHA512(password, salt, tonumber(iters))
    return computed == storedHash
end

serverSettings = function()
    if (fs.exists(filePath.."settings.txt")) then
        local f = fs.open(filePath.."settings.txt", "r")
        local label = f.readLine()
        local oldCurrencyType = nil
        local oldMultiplier = nil

        while label ~= nil do
            if label == "Language:" then
                f.readLine()
            elseif label == "Currency type:" then
                oldCurrencyType = f.readLine()
            elseif label == "Currency value multiplier:" then
                oldMultiplier = tonumber(f.readLine()) or 100
            elseif label == "Currency prices:" then
                local line = f.readLine()
                if line then
                    for k, v in line:gmatch("([^:|]+):([^:|]+)") do
                        if currencyPrices[k] ~= nil then
                            currencyPrices[k] = tonumber(v) or 0
                        end
                    end
                end
            elseif label == "Admin password:" then
                terminalPassword = f.readLine()
                if terminalPassword ~= nil and not tostring(terminalPassword):match("^[^:]+:[^:]+:%d+$") then
                    terminalPassword = srv_hashPassword(tostring(terminalPassword))
                    -- We'll update the file after finishing loading everything
                end
            elseif label == "Cumulative limit:" then
                cumulativeLimit = tonumber(f.readLine()) or 10000
            elseif label == "Limit reach cost:" then
                cumulativePrice = tonumber(f.readLine()) or 100
            elseif label == "Bank owner ID:" then
                bankOwnerKey = tonumber(f.readLine()) or 0
            elseif label == "Network Secret:" then
                networkSecret = tostring(f.readLine() or "")
            elseif label == "Treasury key:" then
                treasuryKey = tonumber(f.readLine() or "0") or 0
            else
                -- Unknown label, try to skip its value
                f.readLine()
            end
            label = f.readLine()
        end
        f.close()

        -- Migration for old format
        if oldCurrencyType and oldMultiplier then
            for k, _ in pairs(currencyPrices) do
                if k == oldCurrencyType then
                    currencyPrices[k] = oldMultiplier
                else
                    currencyPrices[k] = 0
                end
            end
            updateSettingsFile()
        end
    else
        updateSettingsFile()
    end
end

loadClients = function()
    local f = fs.open(filePath.."clientList.txt", "r")
    if (f == nil) then return end
    
    local newData = {}
    while true do
        local line = f.readLine()
        if line == nil or line == "" then break end
        local key = tostring(line or "")
        if key ~= "" then
            local infoPath = filePath.."clientData/"..key.."/info.txt"
            local cf = fs.open(infoPath, "r")
            if cf ~= nil then
                newData[key] = {
                    name = "",
                    balance = 0,
                    color = 1,
                    cumulative = 0,
                    type = "personal",
                    owner = "",
                    treasury = "",
                    password = "",
                    shares = 0,
                    tag = "",
                    frozen = false
                }
                local cline = cf.readLine()
                while cline ~= nil and cline ~= "" do
                    if (cline == "name") then
                        cline = cf.readLine(); newData[key].name = tostring(cline or "")
                    elseif (cline == "balance") then
                        cline = cf.readLine(); newData[key].balance = tonumber(cline or "0") or 0
                    elseif (cline == "color") then
                        cline = cf.readLine(); newData[key].color = tonumber(cline or "1") or 1
                    elseif (cline == "cumulative") then
                        cline = cf.readLine(); newData[key].cumulative = tonumber(cline or "0") or 0
                    elseif (cline == "type") then
                        cline = cf.readLine(); newData[key].type = tostring(cline or "personal")
                    elseif (cline == "owner") then
                        cline = cf.readLine(); newData[key].owner = tostring(cline or "")
                    elseif (cline == "treasury") then
                        cline = cf.readLine(); newData[key].treasury = tostring(cline or "")
                    elseif (cline == "password") then
                        cline = cf.readLine(); newData[key].password = tostring(cline or "")
                    elseif (cline == "shares") then
                        cline = cf.readLine(); newData[key].shares = tonumber(cline or "0") or 0
                    elseif (cline == "tag") then
                        cline = cf.readLine(); newData[key].tag = tostring(cline or "")
                    elseif (cline == "frozen") then
                        cline = cf.readLine(); newData[key].frozen = (tostring(cline or "false") == "true")
                    end
                    cline = cf.readLine()
                end
                cf.close()
            else
                -- Malformed entry; skip safely
                newData[key] = nil
            end
        end
    end
    f.close()
    -- Atomically update the global clientData
    clientData = newData
end

local function getServerData(data)
    local mergedCurrency = {}
    -- Base indices for each currency type (which item the user sets the price for)
    local baseIndices = {
        diamond = 1,
        gold = 2,
        emerald = 1,
        netherite = 1
    }

    for cType, price in pairs(currencyPrices) do
        if price > 0 and currency[cType] then
            local baseIndex = baseIndices[cType] or 1
            local baseRelValue = currency[cType][baseIndex].value
            local multiplier = price / baseRelValue
            
            for _, item in ipairs(currency[cType]) do
                local newItem = {}
                for k, v in pairs(item) do newItem[k] = v end
                newItem.value = item.value * multiplier
                table.insert(mergedCurrency, newItem)
            end
        end
    end
    -- Sort by value ascending so that ATM withdrawal logic (which iterates backwards)
    -- picks the most valuable items first across all types.
    table.sort(mergedCurrency, function(a, b) return a.value < b.value end)

    return true, {
        lang = lang,
        currency = mergedCurrency,
        valueMultiplier = 1,
        terminalPassword = terminalPassword,
        cumulativeLimit = cumulativeLimit,
        cumulativePrice = cumulativePrice,
        sessionNonce = sessionNonce,
        secureEnabled = (networkSecret ~= nil and networkSecret ~= "")
    }
end

updateClientFile = function(key)
    local f = fs.open(filePath.."clientData/"..key.."/info.txt", "w")
    if (f == nil) then return false, STR_ERR_ACCOUNT end

    local acct = clientData[key]
    if not acct then
        f.close()
        return false, STR_ERR_ACCOUNT
    end

    local acctType = tostring(acct.type or "personal")
    f.writeLine("name")
    f.writeLine(tostring(acct.name or ""))
    f.writeLine("balance")
    f.writeLine(tostring(tonumber(acct.balance) or 0))
    f.writeLine("color")
    f.writeLine(tostring(tonumber(acct.color) or 1))
    f.writeLine("type")
    f.writeLine(acctType)
    if acctType ~= "treasury" then
        f.writeLine("cumulative")
        f.writeLine(tostring(tonumber(acct.cumulative) or 0))
        f.writeLine("owner")
        f.writeLine(tostring(acct.owner or ""))
        f.writeLine("treasury")
        f.writeLine(tostring(acct.treasury or ""))
        f.writeLine("password")
        f.writeLine(tostring(acct.password or ""))
        f.writeLine("shares")
        f.writeLine(tostring(tonumber(acct.shares) or 0))
    else
        -- Treasury-specific minimal fields
        f.writeLine("tag")
        f.writeLine(tostring(acct.tag or ""))
        f.writeLine("password")
        f.writeLine(tostring(acct.password or ""))
    end
    f.writeLine("frozen")
    f.writeLine(tostring(acct.frozen and "true" or "false"))
    f.close()

	return true, STR_SUCCESS_UPDATE
end

addCumulative = function(key, amount)
	if (cumulativeLimit == 0) then return nil end
	if (cumulativePrice == 0) then return nil end
	
	local acct = clientData[key]
	if not acct then return end
	
	-- Treasury accounts do not track cumulative transaction volume for fees
	if tostring(acct.type or "") == "treasury" then return end
	
	-- Ensure cumulative is initialized
	acct.cumulative = (tonumber(acct.cumulative) or 0) + amount
	
	if (bankOwnerKey ~= 0) then
		if (key ~= bankOwnerKey) then
			if (clientData[bankOwnerKey] ~= nil and tostring(clientData[bankOwnerKey].type or "") == "treasury") then
				local tax = 0
				local cumulative = acct.cumulative
				while cumulative >= cumulativeLimit do
					tax = tax+cumulativePrice
					cumulative = cumulative-cumulativeLimit
				end
				if (tax > 0) then
					if (tonumber(acct.balance) or 0) >= tax then
						transaction({
							from = key,
							to = bankOwnerKey,
							amount = tax,
							description = STR_EXPENSES
						}, false)
						acct.cumulative = cumulative
					end
				end
			end
		end
	end
end

    if type(os.epoch) == "function" then
        return os.epoch("utc")
    end
    return nil
end
-- Check if an account name already exists (case-insensitive, trimmed)
accountNameExists = function(name)
    local target = tostring(name or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if target == "" then return false end
    local lower = string.lower(target)
    loadClients()
    for _, v in pairs(clientData) do
        local vname = tostring(v.name or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if string.lower(vname) == lower then
            return true
        end
    end
    return false
end

-- Move money from one account to another
transaction = function(data, addToCumulative, skipLoad) --from, to, amount, description
	if (addToCumulative == nil) then addToCumulative = true end
	local from = tostring(data.from)
	local to = tostring(data.to)
	local amount = tonumber(data.amount)
	local description = data.description

	amount = math.ceil(amount*100)/100

    if (amount == 0 or amount == nil) then
        return false, STR_ERR_INVALID_AMOUNT
    end

	if not skipLoad then loadClients() end

    if (from == to) then
        return false, STR_ERR_SAME
    end
    if (clientData[from] == nil) then
        return false, STR_ERR_FROM.." ("..from..")"
    end
    if (clientData[to] == nil) then
        return false, STR_ERR_TO.." ("..to..")"
    end

    -- Enforce transfer rules (only for user-directed transactions):
    -- - Personal accounts can only send to personal accounts
    -- - Company accounts can send to personal or company accounts
    -- - Treasury accounts can send to personal or company accounts
    -- Note: server-initiated flows (e.g., taxes) pass addToCumulative=false and bypass these restrictions
    local fromType = tostring(clientData[from].type or "personal")
    local toType = tostring(clientData[to].type or "personal")
    if (addToCumulative) then
        local allowed = false
        if fromType == "personal" then
            allowed = (toType == "personal" or toType == "treasury")
        elseif fromType == "company" then
            allowed = (toType == "personal" or toType == "company" or toType == "treasury")
        elseif fromType == "treasury" then
            allowed = (toType == "personal" or toType == "company")
        else
            allowed = false
        end
        if not allowed then
            return false, STR_ERR_MIXED_TYPES
        end
    end

	local previousFromBalance = clientData[from].balance
	local newFromBalance = previousFromBalance-amount

    -- Prevent overdrafts for all user-directed transactions
    if (newFromBalance < 0) then
        return false, STR_ERR_NOT_ENOUGH
    end

	local previousToBalance = clientData[to].balance
	local newToBalance = previousToBalance+amount
	local time = getCurrentTime()

	clientData[from].balance = newFromBalance
	clientData[to].balance = newToBalance

	updateClientFile(from)
	updateClientFile(to)

	appendTransactionToLog(from, to, amount, newFromBalance, time,description)
	appendTransactionToLog(to, from, -amount, newToBalance, time, description)

	if (addToCumulative) then
		if not skipLoad then loadClients() end
		addCumulative(from, amount)
		addCumulative(to, amount)
		updateClientFile(from)
		updateClientFile(to)
	end

	return true, STR_SUCCESS_TX
end

deposit = function(data, skipLoad)
	local key = data.key
	local amount = tonumber(data.amount)
	local description = STR_DEPOSIT_DESC

	--print(textutils.serialise(data))
	if (amount == 0 or amount == nil) then
		return false, STR_ERR_INVALID_AMOUNT
	end

	if not skipLoad then loadClients() end

	if (clientData[key] == nil) then
		return false, STR_ERR_FROM.." ("..key..")"
	end

	local previousBalance = clientData[key].balance
	local newBalance = previousBalance+amount

	local time = getCurrentTime()

	clientData[key].balance = newBalance
	updateClientFile(key)

	appendTransactionToLog(key, key, -amount, newBalance, time, description)

	return true, STR_SUCCESS_TX
end

withdraw = function(data, skipLoad)
	local key = data.key
	local amount = tonumber(data.amount)
	local description = STR_WITHDRAW_DESC

	if (amount == 0 or amount == nil) then
		return false, STR_ERR_INVALID_AMOUNT
	end

	if not skipLoad then loadClients() end

	if (key == nil or clientData[key] == nil) then
		return false, STR_ERR_FROM.." ("..tostring(key)..")"
	end

	local previousBalance = clientData[key].balance
	local newBalance = previousBalance-amount

	if (newBalance < 0) then
		return false, STR_ERR_NOT_ENOUGH
	end

	local time = getCurrentTime()
	clientData[key].balance = newBalance
	updateClientFile(key)

	appendTransactionToLog(key, key, amount, newBalance, time, description)

	return true, STR_SUCCESS_TX
end

getTransactionLog = function(data)
	local key = data.key

	local log = {}
	local logCounter = 1
	local f = fs.open(filePath.."clientData/"..key.."/log.txt", "r")
	local line = f.readLine()
	while line ~= nil do
		local lline = line
		local other = ""
		local amount = 0
		local balance = 0
		local time = ""
		local description = ""

		while lline ~= "" and lline ~= nil do
			if (lline == "other") then
				other = f.readLine()
			elseif (lline == "amount") then
				amount = f.readLine()
			elseif (lline == "balance") then
				balance = f.readLine()
			elseif (lline == "time") then
				time = f.readLine()
			elseif (lline == "description") then
				description = f.readLine()
			end
			lline = f.readLine()
		end

		log[logCounter] = {
			other = other,
			amount = amount,
			balance = balance,
			time = time,
			description = description
		}

		logCounter = logCounter+1

		line = f.readLine()
	end

	return true, log
end

-- Create a new client, with its files and folders, given a name, a starting balance and a color
newClient = function(data)
    local name = data.name
    local color = data.color

    -- Enforce global unique account names
    loadClients()
    if accountNameExists(name) then
        return false, "Account name already exists"
    end

	local bankKey = "1000"

	local firstFreeClientNumber = 0
	while firstFreeClientNumber < 9999 do
		firstFreeClientNumber = firstFreeClientNumber+1
		local free = true
		for k, v in pairs(clientData) do
			local clientNumber = tonumber(string.sub(k, 5, 8))
			if (clientNumber == firstFreeClientNumber) then
				free = false
				break
			end
		end
		if (free) then
			break
		end
	end

	local clientNumber = tostring(firstFreeClientNumber)
	clientNumber = string.rep("0", 4-string.len(clientNumber))..clientNumber
	local randomKey = ""
	for i=1, 8 do
		randomKey = randomKey..tostring(math.random(10)-1)
	end
	local key = bankKey..clientNumber..randomKey

	local f = fs.open(filePath.."clientList.txt", "a")
	f.writeLine(key)
	f.close()

	fs.makeDir(filePath.."clientData")
	fs.makeDir(filePath.."clientData/"..key)

	-- Create basic info
    -- Determine treasury link: prefer provided data.treasury if valid; fallback to configured treasuryKey
    local treasuryLink = tostring(data.treasury or "")
    loadClients()
    if treasuryLink ~= "" then
        local t = clientData[treasuryLink]
        if t == nil or tostring(t.type or "") ~= "treasury" then
            treasuryLink = tostring(treasuryKey or "")
        end
    else
        treasuryLink = tostring(treasuryKey or "")
    end

    clientData[key] = {
        name = name,
        balance = 0,
        color = color,
        cumulative = 0,
        type = "personal",
        owner = "",
        treasury = treasuryLink,
        password = "",
        shares = 0
    }
    updateClientFile(key)

	-- Create transaction log
	local f = fs.open(filePath.."clientData/"..key.."/log.txt", "w")
	f.close()

return true, STR_SUCCESS_ACCOUNT
end

-- Create a new Treasury account (server-only) with minimal fields
local function newTreasury(data) -- name, color, tag, password
    local name = data.name
    local color = data.color
    local tag = tostring(data.tag or "")
    local password = tostring(data.password or "")

    loadClients()
    -- Enforce global unique account names
    if accountNameExists(name) then
        return false, "Account name already exists"
    end

    local bankKey = "1000"

    local firstFreeClientNumber = 0
    while firstFreeClientNumber < 9999 do
        firstFreeClientNumber = firstFreeClientNumber+1
        local free = true
        for k, v in pairs(clientData) do
            local clientNumber = tonumber(string.sub(k, 5, 8))
            if (clientNumber == firstFreeClientNumber) then
                free = false
                break
            end
        end
        if (free) then
            break
        end
    end

    local clientNumber = tostring(firstFreeClientNumber)
    clientNumber = string.rep("0", 4-string.len(clientNumber))..clientNumber
    local randomKey = ""
    for i=1, 8 do
        randomKey = randomKey..tostring(math.random(10)-1)
    end
    local key = bankKey..clientNumber..randomKey

    local f = fs.open(filePath.."clientList.txt", "a")
    f.writeLine(key)
    f.close()

    fs.makeDir(filePath.."clientData")
    fs.makeDir(filePath.."clientData/"..key)

    -- Validate tag; fallback to simple derivation if empty
    if tag == "" then
        local words = {}
        for w in tostring(name or ""):gmatch("%S+") do words[#words+1] = w end
        if #words >= 2 then
            tag = (words[1]:sub(1,1) .. words[2]:sub(1,1)):upper()
        else
            local compact = tostring(name or ""):gsub("%s+", "")
            tag = (compact:sub(1,2)):upper()
        end
    else
        tag = tag:upper()
    end

    clientData[key] = {
        name = name,
        balance = 0,
        color = color,
        type = "treasury",
        tag = tag,
        password = (password ~= "" and srv_hashPassword(password)) or ""
    }
    updateClientFile(key)

    local lf = fs.open(filePath.."clientData/"..key.."/log.txt", "w")
    lf.close()

    return true, STR_SUCCESS_ACCOUNT, key
end
-- Set an online password for a client account

local function setPassword(data) -- key, password
    local key = data.key
    local password = tostring(data.password or "")

    loadClients()

    if (key == nil or clientData[key] == nil) then
        return false, STR_ERR_FROM.." ("..tostring(key)..")"
    end

    -- store password as salted hash string
    clientData[key].password = srv_hashPassword(password)
    updateClientFile(key)

    return true, STR_SUCCESS_UPDATE
end

local function toggleAccountFrozen(data) -- key, frozen
    local key = data.key
    local frozen = (data.frozen == true or data.frozen == "true")

    loadClients()

    if (key == nil or clientData[key] == nil) then
        return false, STR_ERR_ACCOUNT
    end

    clientData[key].frozen = frozen
    updateClientFile(key)

    return true, "Account "..(frozen and "frozen" or "unfrozen")
end

-- Change client password with verification
local function changePassword(data) -- key, current, new
    local key = data.key
    local current = tostring(data.current or "")
    local new = tostring(data.new or "")

    loadClients()
    if (key == nil or clientData[key] == nil) then
        return false, STR_ERR_FROM.." ("..tostring(key)..")"
    end
    local stored = clientData[key].password or ""
    if stored == "" then
        return false, "No online account set"
    end
    local isHashed = tostring(stored):match("^[^:]+:[^:]+:%d+$") ~= nil
    if isHashed then
        if not srv_verify(current, stored) then
            return false, "Incorrect password"
        end
    else
        if tostring(current) ~= tostring(stored) then
            return false, "Incorrect password"
        end
    end
    clientData[key].password = srv_hashPassword(new)
    updateClientFile(key)
    return true, STR_SUCCESS_UPDATE
end

-- Card registry helpers: per-account cards tracked in clientData/<key>/cards.txt
local function ensureCardsFile(key)
    local path = filePath.."clientData/"..key.."/cards.txt"
    if not fs.exists(path) then
        local f = fs.open(path, "w")
        if f ~= nil then f.close() end
    end
    return path
end

local function readCards(key)
    local path = ensureCardsFile(key)
    local f = fs.open(path, "r")
    local cards = {}
    if f ~= nil then
        local line = f.readLine()
        while line ~= nil do
            if line ~= "" then
                local id, date, status = line:match("([^|]*)|([^|]*)|([^|]*)")
                if id ~= nil then
                    cards[#cards+1] = { id = id, date = date or "", status = status or "active" }
                end
            end
            line = f.readLine()
        end
        f.close()
    end
    return cards
end

local function writeCards(key, cards)
    local path = ensureCardsFile(key)
    local f = fs.open(path, "w")
    if f == nil then return false end
    for _, c in ipairs(cards) do
        f.writeLine(tostring(c.id).."|"..tostring(c.date).."|"..tostring(c.status))
    end
    f.close()
    return true
end

local function listCards(data) -- key
    local key = tostring(data.key)
    loadClients()
    if clientData[key] == nil then return false, STR_ERR_ACCOUNT end
    -- Treasury is allowed; return all cards
    local cards = readCards(key)
    return true, cards
end

local function registerCard(data) -- key
    local key = tostring(data.key)
    loadClients()
    if clientData[key] == nil then return false, STR_ERR_ACCOUNT end
    local t = tostring(clientData[key].type or "personal")
    local cards = readCards(key)
    -- Enforce active card limits: Treasury=1, Company=2, Personal=unlimited
    local activeCount = 0
    for _, c in ipairs(cards) do if tostring(c.status or "") == "active" then activeCount = activeCount + 1 end end
    if t == "treasury" and activeCount >= 1 then
        return false, "Only 1 active Treasury card allowed"
    elseif t == "company" and activeCount >= 2 then
        return false, "Only 2 active Company cards allowed"
    end
    -- generate unique id
    local id = ""
    repeat
        id = "C"..tostring(math.random(10000000, 99999999))
        local exists = false
        for _, c in ipairs(cards) do if c.id == id then exists = true break end end
        if not exists then break end
    until false
    local date = getCurrentTime()
    cards[#cards+1] = { id = id, date = date, status = "active" }
    writeCards(key, cards)
    return true, id
end

local function cancelCard(data) -- key, id
    local key = tostring(data.key)
    local id = tostring(data.id)
    loadClients()
    if clientData[key] == nil then return false, STR_ERR_ACCOUNT end
    local cards = readCards(key)
    local found = false
    for _, c in ipairs(cards) do
        if tostring(c.id) == id then
            c.status = "canceled"
            found = true
            break
        end
    end
    if not found then return false, STR_ERR_CARD_NOT_FOUND end
    writeCards(key, cards)
    return true, STR_SUCCESS_CARD_CANCELED
end

local function deleteCardRecord(data) -- key, id
    local key = tostring(data.key)
    local id = tostring(data.id)
    loadClients()
    if clientData[key] == nil then return false, STR_ERR_ACCOUNT end
    local cards = readCards(key)
    local newCards = {}
    local found = false
    for _, c in ipairs(cards) do
        if tostring(c.id) == id then
            found = true
        else
            newCards[#newCards+1] = c
        end
    end
    if not found then return false, STR_ERR_CARD_NOT_FOUND end
    writeCards(key, newCards)
    return true, STR_SUCCESS_CARD_DELETED
end

-- Validate that a card exists for an account and is active
local function validateCard(data) -- key, id
    local key = tostring(data.key)
    local id = tostring(data.id)
    loadClients()
    if clientData[key] == nil then return false, STR_ERR_ACCOUNT end
    local cards = readCards(key)
    for _, c in ipairs(cards) do
        if tostring(c.id) == tostring(id) and tostring(c.status or "") == "active" then
            return true, "Card active"
        end
    end
    return false, "Card disabled or not found"
end

-- Secure channel helpers (AES-256-CTR, Base64, keyed MAC using ASHA-like hash)
-- GF(2^8) arithmetic for AES
local function gf_mul(a, b)
    local res = 0
    while b > 0 do
        if bit32.band(b, 1) ~= 0 then res = bit32.bxor(res, a) end
        local hi = bit32.band(a, 0x80)
        a = bit32.band(a * 2, 0xFF)
        if hi ~= 0 then a = bit32.bxor(a, 0x1B) end
        b = bit32.rshift(b, 1)
    end
    return res
end

local function gf_pow(a, e)
    local result = 1
    local base = a
    while e > 0 do
        if bit32.band(e, 1) ~= 0 then result = gf_mul(result, base) end
        base = gf_mul(base, base)
        e = bit32.rshift(e, 1)
    end
    return result
end

local function gf_inv(a)
    if a == 0 then return 0 end
    return gf_pow(a, 254)
end

local function rotl8(b, n)
    return bit32.band(bit32.bor(bit32.lshift(b, n), bit32.rshift(b, 8 - n)), 0xFF)
end

local function sbox_byte(x)
    local inv = gf_inv(x)
    return bit32.band(bit32.bxor(0x63, inv, rotl8(inv,1), rotl8(inv,2), rotl8(inv,3), rotl8(inv,4)), 0xFF)
end

local function rcon(i)
    local c = 1
    if i == 0 then return 0 end
    for _ = 1, i - 1 do
        c = gf_mul(c, 2)
    end
    return c
end

local function sub_word(w)
    return {
        sbox_byte(w[1]), sbox_byte(w[2]), sbox_byte(w[3]), sbox_byte(w[4])
    }
end

local function rot_word(w)
    return { w[2], w[3], w[4], w[1] }
end

local function key_expansion_256(key_bytes)
    local Nk = 8
    local Nb = 4
    local Nr = 14
    local w = {}
    -- first Nk words directly from key
    for i = 0, Nk - 1 do
        w[i] = { key_bytes[i*4+1], key_bytes[i*4+2], key_bytes[i*4+3], key_bytes[i*4+4] }
    end
    for i = Nk, Nb * (Nr + 1) - 1 do
        local temp = { w[i-1][1], w[i-1][2], w[i-1][3], w[i-1][4] }
        if i % Nk == 0 then
            temp = sub_word(rot_word(temp))
            temp[1] = bit32.bxor(temp[1], rcon(i / Nk))
        elseif i % Nk == 4 then
            temp = sub_word(temp)
        end
        w[i] = {
            bit32.bxor(w[i-Nk][1], temp[1]),
            bit32.bxor(w[i-Nk][2], temp[2]),
            bit32.bxor(w[i-Nk][3], temp[3]),
            bit32.bxor(w[i-Nk][4], temp[4])
        }
    end
    -- Flatten round keys
    local round_keys = {}
    for r = 0, Nr do
        local rk = {}
        for c = 0, Nb - 1 do
            local word = w[r*Nb + c]
            rk[#rk+1] = word[1]; rk[#rk+1] = word[2]; rk[#rk+1] = word[3]; rk[#rk+1] = word[4]
        end
        round_keys[r] = rk
    end
    return round_keys
end

local function sub_bytes(state)
    for i = 1, 16 do state[i] = sbox_byte(state[i]) end
end

local function shift_rows(state)
    -- state is column-major: index = r + 4*c
    local t = {}
    for i = 1, 16 do t[i] = state[i] end
    -- row 0 unchanged
    state[1] = t[1];  state[5] = t[5];  state[9]  = t[9];  state[13] = t[13]
    -- row 1 shift left by 1
    state[2] = t[6];  state[6] = t[10]; state[10] = t[14]; state[14] = t[2]
    -- row 2 shift left by 2
    state[3] = t[11]; state[7] = t[15]; state[11] = t[3];  state[15] = t[7]
    -- row 3 shift left by 3
    state[4] = t[16]; state[8] = t[4];  state[12] = t[8];  state[16] = t[12]
end

local function xtime(b)
    local res = bit32.lshift(b, 1)
    if bit32.band(b, 0x80) ~= 0 then res = bit32.bxor(res, 0x1B) end
    return bit32.band(res, 0xFF)
end

local function mix_single_column(s0, s1, s2, s3)
    local t0 = bit32.bxor(xtime(s0), bit32.bxor(bit32.band(s0, 0xFF), xtime(s1), s2, s3))
    local t1 = bit32.bxor(xtime(s1), bit32.bxor(bit32.band(s1, 0xFF), xtime(s2), s3, s0))
    local t2 = bit32.bxor(xtime(s2), bit32.bxor(bit32.band(s2, 0xFF), xtime(s3), s0, s1))
    local t3 = bit32.bxor(xtime(s3), bit32.bxor(bit32.band(s3, 0xFF), xtime(s0), s1, s2))
    return t0, t1, t2, t3
end

local function mix_columns(state)
    for c = 0, 3 do
        local s0 = state[1 + 4*c]
        local s1 = state[2 + 4*c]
        local s2 = state[3 + 4*c]
        local s3 = state[4 + 4*c]
        local t0, t1, t2, t3 = mix_single_column(s0, s1, s2, s3)
        state[1 + 4*c] = bit32.band(t0, 0xFF)
        state[2 + 4*c] = bit32.band(t1, 0xFF)
        state[3 + 4*c] = bit32.band(t2, 0xFF)
        state[4 + 4*c] = bit32.band(t3, 0xFF)
    end
end

local function add_round_key(state, round_key)
    for i = 1, 16 do state[i] = bit32.band(bit32.bxor(state[i], round_key[i]), 0xFF) end
end

local function aes_encrypt_block(input_bytes, round_keys)
    local state = {}
    for i = 1, 16 do state[i] = input_bytes[i] or 0 end
    add_round_key(state, round_keys[0])
    for round = 1, 13 do
        sub_bytes(state)
        shift_rows(state)
        mix_columns(state)
        add_round_key(state, round_keys[round])
    end
    sub_bytes(state)
    shift_rows(state)
    add_round_key(state, round_keys[14])
    return state
end

local function to_bytes(str)
    local t = {}
    for i = 1, #str do t[#t+1] = string.byte(str, i) end
    return t
end

local function from_bytes(t)
    local s = {}
    for i = 1, #t do s[#s+1] = string.char(bit32.band(t[i], 0xFF)) end
    return table.concat(s)
end

local function hex_to_bytes(hex)
    local out = {}
    for i = 1, #hex, 2 do
        local byte = tonumber(hex:sub(i, i+1), 16) or 0
        out[#out+1] = byte
    end
    return out
end

local b64chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'
local function b64encode(data)
    local bytes = to_bytes(data)
    local out = {}
    local i = 1
    while i <= #bytes do
        local b1 = bytes[i] or 0; i = i + 1
        local b2 = bytes[i] or 0; i = i + 1
        local b3 = bytes[i] or 0; i = i + 1
        local n = bit32.bor(bit32.lshift(b1,16), bit32.lshift(b2,8), b3)
        local c1 = bit32.band(bit32.rshift(n,18),0x3F)
        local c2 = bit32.band(bit32.rshift(n,12),0x3F)
        local c3 = bit32.band(bit32.rshift(n,6),0x3F)
        local c4 = bit32.band(n,0x3F)
        local remain = #bytes - (i-1)
        if remain == 1 then
            out[#out+1] = b64chars:sub(c1+1,c1+1)
            out[#out+1] = b64chars:sub(c2+1,c2+1)
            out[#out+1] = '=='
            break
        elseif remain == 2 then
            out[#out+1] = b64chars:sub(c1+1,c1+1)
            out[#out+1] = b64chars:sub(c2+1,c2+1)
            out[#out+1] = b64chars:sub(c3+1,c3+1)
            out[#out+1] = '='
            break
        else
            out[#out+1] = b64chars:sub(c1+1,c1+1)
            out[#out+1] = b64chars:sub(c2+1,c2+1)
            out[#out+1] = b64chars:sub(c3+1,c3+1)
            out[#out+1] = b64chars:sub(c4+1,c4+1)
        end
    end
    return table.concat(out)
end

local function b64decode(data)
    data = data:gsub('%s','')
    local out = {}
    local i = 1
    while i <= #data do
        local c1 = data:sub(i,i); i=i+1
        local c2 = data:sub(i,i); i=i+1
        local c3 = data:sub(i,i); i=i+1
        local c4 = data:sub(i,i); i=i+1
        local n1 = b64chars:find(c1,1,true); local n2 = b64chars:find(c2,1,true)
        local n3 = (c3=='=') and nil or b64chars:find(c3,1,true)
        local n4 = (c4=='=') and nil or b64chars:find(c4,1,true)
        if n1 and n2 then
            local n = bit32.bor(bit32.lshift(n1-1,18), bit32.lshift(n2-1,12), (n3 and bit32.lshift(n3-1,6) or 0), (n4 and (n4-1) or 0))
            out[#out+1] = bit32.band(bit32.rshift(n,16),0xFF)
            if n3 then out[#out+1] = bit32.band(bit32.rshift(n,8),0xFF) end
            if n4 then out[#out+1] = bit32.band(n,0xFF) end
        end
    end
    return from_bytes(out)
end

local function derive_keys()
    -- Derive encryption and MAC material from networkSecret and sessionNonce
    local encHex = srv_ASHA512(networkSecret.."|enc|"..sessionNonce, "", 1)
    local macHex = srv_ASHA512(networkSecret.."|mac|"..sessionNonce, "", 1)
    local encBytes = hex_to_bytes(encHex)
    local aesKey = {}
    for i=1,32 do aesKey[i] = encBytes[i] or 0 end
    local roundKeys = key_expansion_256(aesKey)
    return roundKeys, macHex
end

local function ctr_inc(counter_bytes)
    for i = #counter_bytes, 1, -1 do
        local v = (counter_bytes[i] or 0) + 1
        counter_bytes[i] = bit32.band(v, 0xFF)
        if v <= 0xFF then break end
    end
end

local function aes_ctr_crypt(round_keys, iv_bytes, data)
    local input = to_bytes(data)
    local out = {}
    local counter = {}
    for i=1,16 do counter[i] = iv_bytes[i] or 0 end
    for i = 1, #input, 16 do
        local block = {}
        for j=1,16 do block[j] = input[i+j-1] or 0 end
        local keystream = aes_encrypt_block(counter, round_keys)
        for j=1,16 do out[#out+1] = bit32.band(bit32.bxor(block[j], keystream[j]), 0xFF) end
        ctr_inc(counter)
    end
    return from_bytes(out)
end

local function random_bytes(n)
    local s = {}
    for i=1,n do s[#s+1] = string.char(math.random(0,255)) end
    return table.concat(s)
end

local function keyed_mac(macHex, role, iv_b64, ctr_str, ct_b64)
    local material = role.."|"..iv_b64.."|"..ctr_str.."|"..ct_b64
    return srv_ASHA512(macHex.."|"..material, "", 1)
end

local server_out_ctr = 0
local api_in_ctr = 0

local roundKeys_cached, macHex_cached

local function ensure_keys()
    if (roundKeys_cached == nil or macHex_cached == nil) then
        roundKeys_cached, macHex_cached = derive_keys()
    end
end

local function secure_decrypt(message)
    ensure_keys()
    local iv = b64decode(message.iv or "")
    local ct = b64decode(message.ct or "")
    local mac = tostring(message.mac or "")
    local ctr_str = tostring(message.ctr or "0")
    local expect = keyed_mac(macHex_cached, "api", message.iv, ctr_str, message.ct)
    if expect ~= mac then return false, "Invalid MAC" end
    local iv_bytes = to_bytes(iv)
    local pt = aes_ctr_crypt(roundKeys_cached, iv_bytes, ct)
    if tonumber(ctr_str) <= api_in_ctr then return false, "Replay" end
    api_in_ctr = tonumber(ctr_str)
    return true, pt
end

local function secure_send(sender, payload_table)
    ensure_keys()
    server_out_ctr = server_out_ctr + 1
    local iv = random_bytes(16)
    local iv_b64 = b64encode(iv)
    local payload_str = textutils.serialize(payload_table)
    local ct = aes_ctr_crypt(roundKeys_cached, to_bytes(iv), payload_str)
    local ct_b64 = b64encode(ct)
    local mac = keyed_mac(macHex_cached, "server", iv_b64, tostring(server_out_ctr), ct_b64)
    local envelope = { role = "server", secure = true, iv = iv_b64, ctr = server_out_ctr, ct = ct_b64, mac = mac }
    rednet.send(sender, envelope, "fiducia")
end

function listen()
    while true do
        local ev, p1, p2, p3 = os.pullEventRaw()
        if ev == "terminate" then
            -- ignore Ctrl+T
        elseif ev == "rednet_message" then
            local sender, message, protocol = p1, p2, p3
            if protocol == "fiducia" then
                local isSecure = (networkSecret ~= nil and networkSecret ~= "") and (type(message) == "table" and message.secure == true)
                if isSecure then
                    local ok, pt = secure_decrypt(message)
                    if not ok then
                        secure_send(sender, { success = false, response = pt })
                    else
                        local data = textutils.unserialize(pt) or {}

                        if action == "getClientData" then
                            processSecureRequest(getClientData, sender, data)

function main()
	while true do
		while true do
			local ownerName = STR_NOT_ASSIGNED
			if (tonumber(bankOwnerKey) > 0) then
				if (clientData[bankOwnerKey] == nil) then
					ownerName = STR_DELETED_ACCOUNT
				else
					ownerName = clientData[bankOwnerKey].name
				end
			end
            local command = bankapi.optionMenu(STR_TITLE, {
                {option = "currency_settings",
                text =  STR_CURRENCY_SETTINGS},
                {option = "cumulative_limit",
                text =  STR_CUMULATIVE_LIMIT..": "..cumulativeLimit.."$"},
                {option = "limit_reach_cost",
                text =  STR_LIMIT_REACH_COST..": "..cumulativePrice.."$"},
                {option = "set_owner",
                text =  STR_SET_OWNER..": "..ownerName},
                {option = "establish_treasury",
                text =  "Establish a Treasury"},
            }, 2, 40)

            if (command == "currency_settings") then
                while true do
                    local currencyMenu = {
                        {option = "diamond", text = "Diamond: "..currencyPrices.diamond.."$"},
                        {option = "gold", text = "Gold: "..currencyPrices.gold.."$"},
                        {option = "netherite", text = "Netherite: "..currencyPrices.netherite.."$"},
                        {option = "emerald", text = "Emerald: "..currencyPrices.emerald.."$"},
                        {option = "back", text = "Back"}
                    }
                    local subcommand = bankapi.optionMenu(STR_CURRENCY_SETTINGS, currencyMenu, 2)
                    if subcommand == "back" or subcommand == nil then
                        break
                    else
                        local name = subcommand:gsub("^%l", string.upper)
                        local amount = bankapi.inputNumberScreen({"Set price for "..name, "Set to 0 to disable"}, 1)
                        amount = tonumber(amount)
                        if amount ~= nil and amount >= 0 then
                            currencyPrices[subcommand] = amount
                            updateSettingsFile()
                        end
                    end
                end
			elseif (command == "cumulative_limit") then
				local amount = bankapi.inputNumberScreen({STR_CUMULATIVE_LIMIT_EXPLANATION}, 1)
				amount = tonumber(amount)
				if (amount == nil) then break end
				if (amount <= 0) then bankapi.errorScreen(STR_ERR_INVALID_AMOUNT) end
				cumulativeLimit = amount
				updateSettingsFile()
				break
			elseif (command == "limit_reach_cost") then
				local amount = bankapi.inputNumberScreen({STR_LIMIT_REACH_COST_EXPLANATION}, 1)
				amount = tonumber(amount)
				if (amount == nil) then break end
				if (amount <= 0) then bankapi.errorScreen(STR_ERR_INVALID_AMOUNT) end
				cumulativePrice = amount
				updateSettingsFile()
				break
            elseif (command == "set_owner") then
                -- Limit owner selection to treasury accounts only
                local treasuries = {}
                for k, v in pairs(clientData) do
                    if tostring(v.type or "") == "treasury" then treasuries[k] = v end
                end
                local account = bankapi.selectAccountScreen({STR_SET_OWNER_EXPLANATION}, 1, 0, treasuries)
                if (account == nil) then break end
                if (tonumber(account) > 0) then
                    if clientData[account] ~= nil and tostring(clientData[account].type or "") == "treasury" then
                        bankOwnerKey = account
                        updateSettingsFile()
                    else
                        bankapi.errorScreen("Only treasury accounts can be set as the bank owner")
                    end
                end
            elseif (command == "establish_treasury") then
                local steps = {"Treasury name", "Representative color", "Tag (2-4 letters)", "Treasury password"}
                local name = bankapi.inputTextScreen(steps, 1, 32)
                if (name == nil or name == "") then break end
                local color = bankapi.selectColorScreen(steps, 2)
                if (color == nil) then break end
                local tag = bankapi.inputTextScreen(steps, 3, 4)
                if (tag == nil or tag == "") then break end
                local tpass = bankapi.inputTextScreen(steps, 4, 64)
                if (tpass == nil or tpass == "") then break end
                local ok, msg, key = newTreasury({ name = name, color = color, tag = tag, password = tpass })
                if ok then
                    treasuryKey = key
                    updateSettingsFile()
                    bankapi.successScreen("Treasury established: "..name.." ("..string.upper(tag)..")")
                else
                    bankapi.errorScreen(tostring(msg or "Failed to establish treasury"))
                end
		end
	end
end

end
while true do
    local ok, err = pcall(function()
        parallel.waitForAll(listen, main)
    end)
    if not ok then
        term.setBackgroundColor(colors.black)
        term.setTextColor(colors.red)
        term.clear()
        term.setCursorPos(1,1)
        if err == "Terminated" then
            print("Termination prohibited.")
        else
            print("Critical Server Error: " .. tostring(err))
        end
        print("Restarting in 5 seconds...")
        sleep(5)
    end
end
