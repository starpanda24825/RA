-- ============================================================
-- Regnum Aeternum — Exchange Ticker V1
-- Passive wall-mounted display. No interaction.
-- Scrolls through all active stocks with live prices.
-- Designed for a monitor peripheral on the exchange floor.
-- ============================================================

local API_BASE = "https://YOUR_WORKER_DOMAIN/api/exchange/cc"
local MONITOR_SIDE = "right"  -- Which side the monitor is connected to

-- ── Monitor dimensions ─────────────────────────────────────────
-- Built for a 3-block-tall × 2-block-wide monitor (6 monitor blocks
-- arranged as 2 columns × 3 rows). The code calls setTextScale(0.5),
-- where each block is 14 columns × 10 rows, so the full display is
-- 28 columns × 30 rows.
--
-- To set up:
--   1. Place 6 monitor blocks in a 2-wide × 3-tall rectangle. Adjacent
--      monitor blocks merge into a single multi-block monitor.
--   2. Place this computer touching any block on the MONITOR_SIDE above
--      (default "right").
--
-- Scaling: at text scale 0.5, every extra 16 columns of monitor width
-- adds one more ticker column, and every extra 3 rows adds one more
-- ticker row. A 2×3 monitor shows 9 tickers at a time (1 column × 9
-- rows) and scrolls through the rest over time.
-- ================================================================

local monitor = peripheral.wrap(MONITOR_SIDE)
if not monitor then
  print("No monitor found on " .. MONITOR_SIDE .. " side.")
  return
end

monitor.setTextScale(0.5)
local w, h = monitor.getSize()

local function httpGet(endpoint)
  local url = API_BASE .. endpoint
  local ok, resp = pcall(http.get, url)
  if not ok or not resp then return nil end
  local body = resp.readAll()
  resp.close()
  local ok2, data = pcall(textutils.unserializeJSON, body)
  if not ok2 then return nil end
  return data
end

local function fmt(num, decimals)
  decimals = decimals or 2
  if num == nil then return "---" end
  return string.format("%." .. decimals .. "f", num)
end

local function fmtChg(ch, chp)
  if ch == 0 or ch == nil then return " 0.00" end
  local arrow = ch > 0 and "+" or ""
  return string.format("%s%s %s%%", arrow, fmt(ch), fmt(chp or 0, 1))
end

-- Layout constants
local TICKER_W = 16     -- columns per ticker block (symbol / price / change)
local TICKER_H = 3      -- rows per ticker block
local BANNER_ROWS = 2   -- row 1 = "FDX COMPOSITE", row 2 = index value

local function drawTickerBlock(stock, x, y)
  monitor.setTextColor(colors.yellow)
  monitor.setCursorPos(x, y)
  monitor.write(tostring(stock.t or "---"):sub(1, TICKER_W - 1))

  monitor.setTextColor(colors.white)
  monitor.setCursorPos(x, y + 1)
  monitor.write(fmt(stock.p))

  if stock.ch > 0 then
    monitor.setTextColor(colors.green)
  elseif stock.ch < 0 then
    monitor.setTextColor(colors.red)
  else
    monitor.setTextColor(colors.gray)
  end
  monitor.setCursorPos(x, y + 2)
  monitor.write(fmtChg(stock.ch, stock.chp))
end

-- Draw the visible grid of tickers, starting from startIdx (1-based),
-- wrapping around and scrolling through the full list over time.
local function drawTickers(stocks, startIdx)
  if #stocks == 0 then return end
  local perRow = math.max(1, math.floor(w / TICKER_W))
  local rows = math.max(1, math.floor((h - BANNER_ROWS) / TICKER_H))
  local visible = perRow * rows
  for i = 0, visible - 1 do
    local stock = stocks[((startIdx + i - 1) % #stocks) + 1]
    local col = i % perRow
    local row = math.floor(i / perRow)
    local x = 1 + col * TICKER_W
    local y = BANNER_ROWS + 1 + row * TICKER_H
    drawTickerBlock(stock, x, y)
  end
end

-- Main loop
local scrollOffset = 0
while true do
  local idxData = httpGet("/index")
  local marketData = httpGet("/market")

  monitor.setBackgroundColor(colors.black)
  monitor.clear()

  -- Index banner
  monitor.setCursorPos(1, 1)
  monitor.setTextColor(colors.gray)
  monitor.write("FDX COMPOSITE")
  monitor.setCursorPos(1, 2)
  if idxData and idxData.success then
    local idx = idxData.response
    monitor.setTextColor(colors.yellow)
    monitor.write(fmt(idx.idx))
    monitor.setTextColor(colors.gray)
    monitor.write("  A" .. idx.adv .. " D" .. idx.dec .. " U" .. idx.unc)
  else
    monitor.setTextColor(colors.white)
    monitor.write("----")
  end

  -- Ticker grid
  if marketData and marketData.success then
    local stocks = marketData.response or {}
    drawTickers(stocks, scrollOffset + 1)
    if #stocks > 0 then
      scrollOffset = (scrollOffset + 1) % #stocks
    end
  end

  sleep(30)  -- Refresh every 30 seconds
end
