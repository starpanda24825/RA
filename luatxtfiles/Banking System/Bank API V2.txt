local BASE_URL = "https://regnumaeternum.starpanda24825.workers.dev"

local cc_token = ""
if fs.exists("cc_token.txt") then
    local tf = fs.open("cc_token.txt", "r")
    if tf then
        cc_token = tf.readAll():gsub("^%s+", ""):gsub("%s+$", "")
        tf.close()
    end
end

local function httpPost(path, body)
    local url = BASE_URL .. path
    local headers = {
        ["Content-Type"] = "application/json",
    }
    if cc_token ~= "" then
        headers["Authorization"] = "Bearer " .. cc_token
    end
    local bodyStr = textutils.serializeJSON(body)
    local ok, result = pcall(function()
        local req = http.post(url, bodyStr, headers)
        if req then
            local raw = req.readAll()
            req.close()
            return raw
        end
        return nil
    end)
    if not ok then
        return nil
    end
    if result == nil then
        return nil
    end
    local ok2, decoded = pcall(textutils.unserializeJSON, result)
    if ok2 and decoded then
        return decoded
    end
    return nil
end

local function httpGet(path)
    local url = BASE_URL .. path
    local headers = {}
    if cc_token ~= "" then
        headers["Authorization"] = "Bearer " .. cc_token
    end
    local ok, err = pcall(function()
        local req = http.get(url, headers)
        if req then
            local raw = req.readAll()
            req.close()
            return raw
        end
        return nil
    end)
    if not ok or err == nil then
        return nil
    end
    local ok2, decoded = pcall(textutils.unserializeJSON, err)
    if ok2 and decoded then
        return decoded
    end
    return nil
end

local serverData = nil

-- Action-to-endpoint mapping for sendRequest
local ACTION_ENDPOINTS = {
    getClientData      = "/api/banking/cc/client-data",
    transaction        = "/api/banking/cc/transaction",
    deposit            = "/api/banking/cc/deposit",
    withdraw           = "/api/banking/cc/withdraw",
    getTransactionLog  = "/api/banking/cc/transaction-log",
    ["new"]            = "/api/banking/cc/new-account",
    newCompany         = "/api/banking/cc/new-company",
    ["delete"]         = "/api/banking/cc/delete-account",
    login              = "/api/banking/cc/login",
    treasuryLogin      = "/api/banking/cc/treasury-login",
    resetPassword      = "/api/banking/cc/reset-password",
    validateCard       = "/api/banking/cc/validate-card",
    registerCard       = "/api/banking/cc/register-card",
    listCards          = "/api/banking/cc/list-cards",
    cancelCard         = "/api/banking/cc/cancel-card",
    freezeAccount      = "/api/banking/cc/freeze-account",
}

function sendRequest(message)
    local action = message.action
    local endpoint = ACTION_ENDPOINTS[action]
    if endpoint == nil then
        return { success = false, response = "Unknown action: " .. tostring(action) }
    end
    local body = {}
    for k, v in pairs(message) do
        if k ~= "action" then body[k] = v end
    end
    local result = httpPost(endpoint, body)
    if result == nil then
        return { success = false, response = "Connection failed" }
    end
    -- Log unknown actions to help debugging
    if result.success == false and result.response and string.find(tostring(result.response), "Unknown") then
        print("Warning: " .. tostring(result.response))
    end
    return result
end

function getServerData()
    print("Connecting to web server...")
    local result = httpPost("/api/banking/cc/server-data", {})
    if result == nil then
        error("Could not connect to web server at " .. BASE_URL)
    end
    if not result.success then
        error("Server error: " .. tostring(result.response))
    end
    serverData = result.response or {}
    print("Successfully connected to web server")
    return serverData
end

function getClientData()
    local message = {
        action = "getClientData"
    }
    for i = 1, 5 do
        local resp = sendRequest(message)
        if resp and resp.success and resp.response ~= nil then
            return resp.response
        end
        sleep(0.2)
    end
    return {}
end

function drawButton(primary, secondary, textColor, x, y, w, text)
	local ch = string.char(127)
	term.setCursorPos(x, y)
	term.setBackgroundColor(secondary)
	term.setTextColor(primary)
	term.write(" ")
	term.write(ch)
	term.setBackgroundColor(primary)
	term.setTextColor(secondary)
	term.write(ch)
	term.setBackgroundColor(primary)
	term.write(string.rep(" ", w-6))
	term.setBackgroundColor(primary)
	term.setTextColor(secondary)
	term.write(ch)
	term.setBackgroundColor(secondary)
	term.setTextColor(primary)
	term.write(ch)
	term.write(" ")

	-- Shadow
	term.setBackgroundColor(backgroundColor)
	term.setTextColor(colors.black)
	term.write(string.char(148))
	local scrW, scrH = term.getSize()
	if (y < scrH) then
		term.setCursorPos(x, y+1)
		term.write(string.char(130))
		term.write(string.rep(string.char(131), w-1))
		term.write(string.char(129))
	end

	term.setBackgroundColor(primary)
	term.setTextColor(textColor)

	term.setCursorPos(x+w/2-string.len(text)/2, y)
	term.write(text)

	return {x=x, y=y, w=w}
end

function newAccount(name, balance, color, treasuryKey)
    local message = {
        action = "new",
        name = name,

        color = color

local function startRead(maxLength)
	local scrW, scrH = term.getSize()
	readingPosX, readingPosY = term.getCursorPos()
	reading = true
	readingString = ""
	readingMax = maxLength
	term.setBackgroundColor(colors.white)
	term.setTextColor(colors.black)
	term.setCursorPos(readingPosX, readingPosY)
	term.write(string.rep(" ", scrW-2))
	term.setCursorBlink(true)
end

local function processChar(char)
	term.setBackgroundColor(colors.white)
	term.setTextColor(colors.black)
	term.setCursorPos(readingPosX, readingPosY)
	readingString = readingString..char
	term.write(readingString)
end

local function processKey(key)
	if (key == keys.backspace) then
		readingString = string.sub(readingString, 1, string.len(readingString)-1)
		term.setBackgroundColor(colors.white)
		term.setCursorPos(readingPosX+string.len(readingString), readingPosY)
		term.write(" ")
		term.setCursorPos(readingPosX+string.len(readingString), readingPosY)
	elseif (key == keys.enter) then
		term.setCursorBlink(false)
		return readingString
	end
	return nil
end

local function cancelableRead(maxLength, secondaryButtonX)
	startRead(maxLength)
	local back = drawBackButton()
	local accept = drawAcceptButton()
	term.setCursorPos(readingPosX, readingPosY)
	local input = nil
	while input == nil do
		local event, a, b, c = safePullEvent()
		if (event == "char") then
			processChar(a)
		elseif (event == "key") then
			input = processKey(a)
		elseif (event == "mouse_click") then
			local cx = b
			local cy = c
			if (mouseInButton(back, cx, cy)) then
				term.setCursorBlink(false)
				return nil
			elseif (mouseInButton(accept, cx, cy)) then
				term.setCursorBlink(false)
				input = readingString
				break
			elseif (secondaryButtonX ~= nil) then
				if (cy == back.y and cx >= secondaryButtonX) then
					return true
				end
			end
		end
	end
	term.setCursorBlink(false)
	if (readingMax > 0 and string.len(input) > readingMax) then
		input = string.sub(input, 1, readingMax)
	end
	return input
end

function transactionLogScreen(key)
	local tempClientData = getClientData()
	local backwardsLogs = getTransactionLog(key)
	local allLogs = {}

	local logCount = #backwardsLogs
	for i=0, logCount-1 do
		allLogs[i+1] = backwardsLogs[logCount-i] -- Newest first
	end

	local scrW, scrH = term.getSize()
	local y = 3 -- Moved down for search bar
	local floor = 1
	local x = 1
	local w = scrW

	if (pocket) then
		floor = 2
	end

	local first = 0
	local logHeight = 2
	local max = math.floor((scrH-floor-logHeight-1)/logHeight)

	local scrollButtonY = scrH-1
	local scrollButtonCenterX = math.floor(scrW/2)+10

	if (pocket) then
		scrollButtonY = scrH-3
		scrollButtonCenterX = math.floor(scrW/2)+1
	end

	local filterText = ""

	while true do
		local filteredLogs = {}
		if filterText == "" then
			filteredLogs = allLogs
		else
			local lowerFilter = string.lower(filterText)
			for _, v in ipairs(allLogs) do
				local name = "deleted"
				if tempClientData[v.other] then name = string.lower(tempClientData[v.other].name) end
				local desc = string.lower(v.description or "")
				local date = string.lower(v.time or "")
				if string.find(name, lowerFilter) or string.find(desc, lowerFilter) or string.find(date, lowerFilter) then
					table.insert(filteredLogs, v)
				end
			end
		end

		drawBackground()
		
		-- Search Bar
		term.setCursorPos(1, 1)
		term.setBackgroundColor(colors.white)
		term.setTextColor(colors.gray)
		term.write(" Search: ")
		term.setTextColor(colors.black)
		local displayFilter = filterText
		if filterText == "" then 
			term.setTextColor(colors.lightGray)
			displayFilter = "Type to filter (Name/Desc/Date)" 
		end
		term.write(string.sub(displayFilter, 1, scrW - 10))
		term.write(string.rep(" ", scrW - string.len(" Search: ") - string.len(string.sub(displayFilter, 1, scrW - 10))))

		local prevPage = drawButton(buttonColor, secondaryButtonColor, buttonTextColor,scrollButtonCenterX-13, scrollButtonY, 7, string.char(27))
		if (first <= 0) then
    		drawButton(colors.lightGray, colors.gray, buttonTextColor, scrollButtonCenterX-13, scrollButtonY, 7, string.char(27))
		end

		local totalLogs = #filteredLogs
		local i = 0
		local buttons = {}
		
		for k = first + 1, math.min(first + max, totalLogs) do
			local v = filteredLogs[k]
			local order = totalLogs - k + 1
			local amountText = ""
			if (tonumber(v.amount) > 0) then
				amountText = "+$"..v.amount
			else
				amountText = "-$"..math.abs(v.amount)
			end

			-- content row
			term.setCursorPos(1, y + i)
			buttons[y + i] = k
			term.setBackgroundColor(colors.white)
			term.write(string.rep(" ", scrW))
			
			term.setCursorPos(1, y + i)
			local name = tempClientData[v.other] and tempClientData[v.other].name or "<deleted>"
			
			term.setTextColor(colors.lightGray)
			term.write(string.sub(v.time, 1, 5).." ") -- Time prefix

			term.setTextColor(colors.black)
			term.write(string.sub(name, 1, math.floor(scrW*0.3)))

			term.setCursorPos(scrW*0.4, y + i)
			if (tonumber(v.amount) >= 0) then
				term.setTextColor(colors.green)
			else
				term.setTextColor(colors.red)
			end
			term.write(amountText)

			term.setCursorPos(scrW*0.6, y + i)
			term.setTextColor(colors.gray)
			term.write(string.sub(v.description, 1, scrW - (scrW*0.6) - 1))

			-- separator
			if k < math.min(first + max, totalLogs) then
				term.setCursorPos(1, y + i + 1)
				term.setBackgroundColor(backgroundColor)
				term.setTextColor(colors.lightGray)
				term.write(string.rep(string.char(140), scrW))
			end

			i = i + logHeight
		end

		local nextPage = drawButton(buttonColor, secondaryButtonColor, buttonTextColor, scrollButtonCenterX+5, scrollButtonY, 7, string.char(26))
		if (first+max >= totalLogs) then
    		drawButton(colors.lightGray, colors.gray, buttonTextColor, scrollButtonCenterX+5, scrollButtonY, 7, string.char(26))
		end

		-- page indicator
		local pages = math.max(1, math.ceil(totalLogs/max))
		local pageText = tostring(math.floor(first/max)+1).."/"..pages
		term.setCursorPos(scrollButtonCenterX-string.len(pageText)/2, scrollButtonY)
		term.setBackgroundColor(backgroundColor)
		term.setTextColor(colors.orange)
		term.write(pageText)

		local backButton = drawBackButton()

		local eventData = {safePullEvent()}
		local event = eventData[1]

		if event == "mouse_click" then
			local cx = eventData[3]
			local cy = eventData[4]
			
			if cy == 1 then -- Clicked search bar
				term.setCursorPos(10, 1)
				term.setCursorBlink(true)
				local input = cancelableRead(30, nil)
				if input ~= nil then
					filterText = input
					first = 0
				end
				term.setCursorBlink(false)
			elseif (mouseInButton(prevPage, cx, cy)) then
				if (first > 0) then
					first = first-max
				end
			elseif (mouseInButton(nextPage, cx, cy)) then
				if (first+max < totalLogs) then
					first = first+max
				end
			elseif cy >= y and buttons[cy] then
				transactionInfoScreen(filteredLogs[buttons[cy]])
			elseif (mouseInButton(backButton, cx, cy)) then
				return nil
			end
		elseif event == "mouse_scroll" then
			local scroll = eventData[2]
			if (scroll < 0) then
				if (first > 0) then first = first-max end
			else
				if (first+max < totalLogs) then first = first+max end
			end
		elseif event == "char" then
			filterText = filterText .. eventData[2]
			first = 0
		elseif event == "key" then
			if eventData[2] == keys.backspace then
				filterText = string.sub(filterText, 1, -2)
				first = 0
			end
		end
	end
end

function optionMenu(title, options, spacing, width)
	if (spacing == nil) then spacing = 2 end
	if (width == nil) then width = 30 end
	drawBackground()
	local buttons = {}
	local scrW, scrH = term.getSize()
	local w = width
	local x = scrW/2-w/2+1
	local y = math.floor(scrH/2-(#options+1)*spacing/2)+2
	local i = 0
	term.setCursorPos(scrW/2-string.len(title)/2+1, y+i)
	term.setTextColor(specialTextColor)
	term.write(title)
	i = i+spacing

	term.setBackgroundColor(buttonColor)
	term.setTextColor(buttonTextColor)
	for k, v in ipairs(options) do
		drawButton(buttonColor, secondaryButtonColor, buttonTextColor, x, y+i, w, v.text)
		
		buttons[y+i] = v.option
		i = i+spacing
	end

	while true do
		local event, button, cx, cy = safePullEvent("mouse_click")
		if (cx >= x-1 and cx <= x+width-1 and cy >= y and cy < y+(#options+1)*spacing) then
			if buttons[cy] ~= nil then
				return buttons[cy]
			end
		end
	end
end

end

function selectAccountScreen(steps, currentStep, disabledAccount, overrideClientData)
	local tempClientData
	
	if (overrideClientData == nil) then
		tempClientData = getClientData()
	else
		tempClientData = overrideClientData
	end

	local clientCount = 0
	for k, v in pairs(tempClientData) do
		clientCount = clientCount+1
	end

    if (clientCount == 0) then
        errorScreen("There are no accounts at the moment")
        return nil
    end
	
	local scrW, scrH = term.getSize()
	local x, y, w, h = scrW/4, #steps+4, scrW/2+2, scrH-5

	local back = drawBackButton()

	local first = 0
	local max = scrH-y-2

	local upButtonY = y-2
	local downButtonY = y+1+max

	while true do
		drawSteps(steps, currentStep)
		drawBackButton()
		local i = 0
		local buttons = {}

		if (first > 0) then
			term.setCursorPos(scrW/2-3,upButtonY)
			term.setTextColor(buttonTextColor)
			term.setBackgroundColor(buttonColor)
			term.write("  "..string.char(24).."  ")
		end

		local totalAccounts = 0
		local showAccounts = 0
		for k, v in pairs(tempClientData) do
			if (disabledAccount ~= nil and disabledAccount ~= k) then
				if (totalAccounts >= first and showAccounts < first+max) then
					buttons[y+i] = k
					term.setCursorPos(x,y+i)
					term.setTextColor(contrastColor(tonumber(v.color)))
					term.setBackgroundColor(tonumber(v.color))
					term.write(string.rep(" ", w))
					term.setCursorPos(x+w/2-string.len(v.name)/2-1,y+i)
					term.write(v.name)
					i = i+1
				end
				showAccounts = showAccounts+1
			end
			totalAccounts = totalAccounts+1
		end

			term.setCursorPos(scrW/2-3,downButtonY)
			term.setTextColor(buttonTextColor)
			term.setBackgroundColor(buttonColor)
			term.write("  "..string.char(25).."  ")

			end

			return buttons[cy]

		end

function selectColorScreen(steps, currentStep)
    drawSteps(steps, currentStep)
    local y = #steps+3
    local scrW, scrH = term.getSize()
    local x = math.floor(scrW/2 - 20)
    local w = 5 -- swatch width
    local h = 1 -- swatch height

    -- Explicit palette and names (16 ComputerCraft colors)
    local palette = {
        colors.white, colors.orange, colors.magenta, colors.lightBlue,
        colors.yellow, colors.lime, colors.pink, colors.gray,
        colors.lightGray, colors.cyan, colors.purple, colors.blue,
        colors.brown, colors.green, colors.red, colors.black
    }
    local names = {
        "White","Orange","Magenta","Light Blue",
        "Yellow","Lime","Pink","Gray",
        "Light Gray","Cyan","Purple","Blue",
        "Brown","Green","Red","Black"
    }
    local buttons = {}
    for _x = 0, 7 do buttons[_x] = {} end

    -- Draw 8x2 grid to expose all colors with consistent ordering
    for i = 1, #palette do
        local col = palette[i]
        local gx = (_G.math.floor((i-1) % 8))
        local gy = (_G.math.floor((i-1) / 8))
        local bx = x + gx * w + 1
        local by = y + gy * (h + 1)
        term.setCursorPos(bx, by)
        term.setBackgroundColor(col)
        term.setTextColor(contrastColor(col))
        term.write("[")
        term.write(string.rep(" ", w-2))
        term.write("]")
        buttons[gx][gy] = col
    end

    -- Draw names column as a reference
    local namesX = x + (8 * w) + 3
    term.setBackgroundColor(colors.black)
    term.setTextColor(colors.white)
    for i = 1, #names do
        local gy = _G.math.floor((i-1) / 8)
        local rowY = y + gy * (h + 1)
        term.setCursorPos(namesX, rowY)
        local label = table.concat(names, ", ", gy*8+1, _G.math.min(gy*8+8, #names))
        term.write(label)
    end

    local backButton = drawBackButton()

    while true do
        local event, button, cx, cy = safePullEvent("mouse_click")
        if (cx >= x and cx <= x + w*8 and cy >= y and cy < y + 2*(h+1)) then
            local _x = _G.math.floor((cx - x) / w)
            local dy = _G.math.floor((cy - y) / (h + 1))
            if buttons[_x] and buttons[_x][dy] then
                return buttons[_x][dy]
            end
        end
        if mouseInButton(backButton, cx, cy) then
            return nil
        end
    end
end

function errorScreen(message)
    term.setBackgroundColor(colors.red)
    term.setTextColor(colors.white)

    term.setBackgroundColor(colors.black)
    local scrW, scrH = term.getSize()

    if (type(message) == 'table') then
        local count = #message
        for k, v in pairs(message) do
            local s = tostring(v or "")
            term.setCursorPos(scrW/2-string.len(s)/2+1, scrH/2-math.ceil(count/2)+k)
            term.write(s)
        end
        local tid = os.startTimer(1.5 * #message)
        while true do
            local ev, p1 = os.pullEventRaw()
            if ev == "terminate" then
                -- ignore Ctrl+T
            elseif ev == "timer" and p1 == tid then
                break
            end
        end
    else
        local s = tostring(message or "Error")
        term.setCursorPos(scrW/2-string.len(s)/2+1, scrH/2)
        term.write(s)
        local tid = os.startTimer(1.5)

                -- ignore Ctrl+T
            elseif ev == "timer" and p1 == tid then
                break
            end
        end
    end
end

-- In-game password login screen. Accepts an account key or the account's
-- name (username), verifies the password server-side, and returns the
-- account key + name on success (nil on failure/cancel).
function loginScreen()
    drawBackground()
    local scrW, scrH = term.getSize()
    term.setTextColor(colors.white)
    term.setCursorPos(scrW/2 - string.len("Log in with password")/2 + 1, 2)
    term.write("Log in with password")

    term.setTextColor(colors.white)
    term.setCursorPos(3, 4)
    term.write("Account key or username:")
    term.setCursorPos(3, 5)
    term.setTextColor(colors.yellow)
    local identifier = tostring(read() or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if identifier == "" then return nil end

    term.setTextColor(colors.white)
    term.setCursorPos(3, 7)
    term.write("Password:")
    term.setCursorPos(3, 8)
    term.setTextColor(colors.yellow)
    local password = readPassword(64)
    if password == "" then return nil end

    local ok, resp = login(identifier, password)
    if ok then
        return resp.key, resp.name
    else
        errorScreen(tostring(resp))
        return nil
    end
end

-- Re-validate a card before a sensitive operation (transfer/deposit/withdraw/
-- purchase). Pass id = nil/"" for password logins (no card). Returns false if
-- the card was cancelled or the account was frozen server-side mid-session.
-- Active card tracking for this session. Clients call setCard after a
-- successful card validation and clearCard on logout/password login. The
-- transaction/deposit/withdraw functions re-validate this card so a card
-- cancelled server-side mid-session stops working immediately.
function setCard(key, id)
    currentCardKey = key
    currentCardId = id
end

function clearCard()
    currentCardKey = nil
    currentCardId = nil
end

function revalidateCard(key, id)
    if id == nil or id == "" then
        return true -- password login: frozen state is still enforced server-side
    end
    local ok = validateCard(key, id)
    return ok
end

end

-- Security hardening: ignore Ctrl+T terminate and only emit desired events
function safePullEvent(filter)
    while true do
        local ev = {os.pullEventRaw()}
        local e = ev[1]
        if e ~= "terminate" then
            if filter == nil or e == filter then
                return table.unpack(ev)
            end
        end
        -- otherwise, keep waiting
    end
end

-- Masked password input that ignores Ctrl+T terminate events
function readPassword(maxLength)
    if maxLength == nil then maxLength = 0 end
    local s = ""
    term.setCursorBlink(true)
    while true do
        local event, a = safePullEvent()
        if event == "char" then
            if maxLength == 0 or #s < maxLength then
                s = s .. a
                term.write("*")
            end
        elseif event == "key" then
            if a == keys.backspace then
                if #s > 0 then
                    s = s:sub(1, #s-1)
                    local x,y = term.getCursorPos()
                    term.setCursorPos(x-1, y)
                    term.write(" ")
                    term.setCursorPos(x-1, y)
                end
            elseif a == keys.enter then
                term.setCursorBlink(false)
                return s
            end
        end
    end
end
