-- ============================================================
-- Sublevel Cannon Computer — Regnum Aeternum
-- ComputerCraft bridge for MOBILE / SUBLEVEL cannons (Create: Big Cannons)
--
-- ────────────────────────────────────────────────────────────
-- IN-GAME SETUP (read this first)
-- ────────────────────────────────────────────────────────────
--
-- REQUIRED MODS
--   1. Create: Big Cannons (CBC)
--   2. CC: Tweaked
--   3. Create Big Cannons: Peripherals (cbcperipheral)
--      This is what lets a CC computer talk to a cannon mount.
--
-- REQUIRED EXTRAS
--   - A wireless modem on this computer, and a GPS host on the
--     network. The world is already set up so that gps.locate(2)
--     returns this computer's coordinates — that is how the website
--     knows where the moving cannon is on the map.
--   - For ship heading: TWO more computers running the "Sublevel Ship
--     GPS" program — one at the FRONT (bow) and one at the BACK
--     (stern), as far apart as possible along the fore-aft centerline.
--     This cannon computer listens to both and derives the ship's
--     orientation.
--
-- BUILD
--   1. Build your cannon on the ship's cannon mount (NOT the fixed
--      mount — the fixed mount can't be aimed by a computer).
--   2. Place an advanced Computer next to (touching) the cannon
--      mount block so the computer can see the mount peripheral.
--   3. Wire the cannon's firing mechanism. Two options:
--        a) Peripheral fire (recommended) — cbcperipheral's fire()
--           calls the cannon directly.
--        b) Redstone fire — run a redstone cable from a side of the
--           computer to the firing block and set fireMode=redstone
--           in subcannon_cfg.txt.
--   4. Make sure the cannon is LOADED (shell + powder) before firing.
--
-- RUN
--   1. Put this program on the computer as "startup" (it is a daemon).
--   2. First boot prompts for: cannon length (blocks) and the
--      direction the cannon faces when assembled. Saved to
--      subcannon_cfg.txt. This cannon is always reported as sublevel.
--   3. The computer registers itself with the website as a SUBLEVEL
--      cannon. It shows "Awaiting approval" until an officer accepts
--      it in the Ballistic Calculator → Cannon Registry tab.
--   4. Once accepted it does two jobs at the same time:
--        a) Polls for fire commands (yaw + pitch). On a command it
--           runs the same sequence as static cannons:
--              disassemble → assemble → aim → fire → disassemble
--        b) A concurrent GPS worker keeps calling gps.locate() and the
--           main loop pushes the freshest coordinates to the website
--           roughly every 3 seconds, so its dot on the map follows the
--           cannon as the ship moves. Until the first real fix is found
--           it never reports a position, so the dot never snaps to 0,0,0.
--
-- DIRECTION FACED WHEN ASSEMBLED ("facing")
--   Same convention as the static Cannon Computer:
--      0   = South (+Z)
--      90  = West
--      180 = North
--      270 = East
--   Find it with F3: face the direction your barrel points when the
--   cannon is assembled, and read the "Facing" value. Calibrate with
--   yawOffset / pitchScale in subcannon_cfg.txt (see CALIBRATION).
--
-- CALIBRATION
--   CBC's yaw convention should match Minecraft (0 = south, positive
--   clockwise). If a fire command turns the cannon the wrong way or is
--   offset, tune these in subcannon_cfg.txt:
--     yawOffset  — added to the incoming yaw (degrees)
--     pitchScale — multiplied by the incoming pitch (1 or -1)
-- ============================================================

-- ------------------------------------------------------------
-- Configuration
-- ------------------------------------------------------------
local BASE_URL = "https://regnumaeternum.starpanda24825.workers.dev"
local POLL_PATH = "/api/ballistics/cc/poll"

-- Mount peripheral type from cbcperipheral.
local MOUNT_TYPE = "cbc_cannon_mount"

-- How long (seconds) to let gps.locate run per fix attempt. GPS runs in a
-- concurrent worker (see bottom of file), so this blocks that worker — not
-- the poll / fire loop.
local GPS_TIMEOUT = 2

-- Sequence timing (seconds) — identical to the static cannon.
local SEQ_DISASSEMBLE = 0.5
local SEQ_ASSEMBLE    = 1.0
local SEQ_BEFORE_AIM  = 4.0   -- wait after assembling, before aiming
local SEQ_BEFORE_FIRE = 5.0   -- wait after aiming, before firing
local SEQ_AFTER_FIRE  = 2.0   -- wait after firing, before disassembling
local AIM_TIMEOUT     = 8.0   -- max seconds to wait for aim to settle
local AIM_TOLERANCE   = 0.5   -- degrees

-- Config file paths (kept separate from the static cannon's files).
local CFG_PATH = "subcannon_cfg.txt"
local ID_PATH  = "subcannon_id.txt"

-- ------------------------------------------------------------
-- Config defaults
-- ------------------------------------------------------------
local cfg = {
  length       = 4,
  facing       = 0,
  message      = "",
  pollInterval = 1,      -- seconds after each poll; total loop ≈ 3s
  yawOffset    = 0,
  pitchScale   = 1,
  shipRelativeYaw = true,  -- mount yaw is relative to the ship (normal for
                           -- Create contraptions). Set false if getYaw/setYaw
                           -- are already in absolute world coordinates.
  fireMode     = "peripheral",  -- "peripheral" or "redstone"
  redstoneSide = "",            -- e.g. "left", "right", "back", "top"
  firePulse    = 0.5,           -- redstone pulse length (seconds)
}

-- ------------------------------------------------------------
-- Small helpers
-- ------------------------------------------------------------
local function trim(s)
  s = tostring(s or "")
  s = s:gsub("^%s+", "")
  s = s:gsub("%s+$", "")
  return s
end

local function randomHex(len)
  local chars = "0123456789abcdef"
  local s = ""
  for i = 1, len do
    local r = math.random(1, #chars)
    s = s .. chars:sub(r, r)
  end
  return s
end

local function readConfig()
  if not fs.exists(CFG_PATH) then return nil end
  local f = fs.open(CFG_PATH, "r")
  if not f then return nil end
  local loaded = {}
  while true do
    local line = f.readLine()
    if line == nil then break end
    local k, v = line:match("^([%w_]+)%s*=%s*(.-)%s*$")
    if k then loaded[k] = v end
  end
  f.close()
  return loaded
end

local function writeConfig()
  local f = fs.open(CFG_PATH, "w")
  f.writeLine("length=" .. tostring(cfg.length))
  f.writeLine("facing=" .. tostring(cfg.facing))
  f.writeLine("message=" .. tostring(cfg.message))
  f.writeLine("pollInterval=" .. tostring(cfg.pollInterval))
  f.writeLine("yawOffset=" .. tostring(cfg.yawOffset))
  f.writeLine("pitchScale=" .. tostring(cfg.pitchScale))
  f.writeLine("shipRelativeYaw=" .. tostring(cfg.shipRelativeYaw))
  f.writeLine("fireMode=" .. tostring(cfg.fireMode))
  f.writeLine("redstoneSide=" .. tostring(cfg.redstoneSide))
  f.writeLine("firePulse=" .. tostring(cfg.firePulse))
  f.close()
end

local function loadConfig()
  local loaded = readConfig()
  if loaded then
    cfg.length       = tonumber(loaded.length) or cfg.length
    cfg.facing       = tonumber(loaded.facing) or cfg.facing
    cfg.message      = loaded.message or cfg.message
    cfg.pollInterval = tonumber(loaded.pollInterval) or cfg.pollInterval
    cfg.yawOffset    = tonumber(loaded.yawOffset) or cfg.yawOffset
    cfg.pitchScale   = tonumber(loaded.pitchScale) or cfg.pitchScale
    if loaded.shipRelativeYaw ~= nil then
      cfg.shipRelativeYaw = (loaded.shipRelativeYaw ~= "false")
    end
    cfg.fireMode     = loaded.fireMode or cfg.fireMode
    cfg.redstoneSide = loaded.redstoneSide or cfg.redstoneSide
    cfg.firePulse    = tonumber(loaded.firePulse) or cfg.firePulse
  end
end

local function readLine(prompt)
  write(prompt)
  return trim(read())
end

-- First-boot interactive setup.
local function setup()
  term.clear()
  term.setCursorPos(1, 1)
  print("=== Sublevel Cannon Computer — first-time setup ===")
  print("")

  local len = tonumber(readLine("Cannon length in blocks (e.g. 4): "))
  cfg.length = len and math.floor(len) or cfg.length

  local face = tonumber(readLine("Direction faced when assembled (0=S,90=W,180=N,270=E): "))
  cfg.facing = face or cfg.facing

  print("")
  print("Message shown to the officer reviewing your request.")
  cfg.message = readLine("Message (leave blank for none): ")

  writeConfig()
  print("")
  print("Saved " .. CFG_PATH .. ". Starting…")
  sleep(1)
end

-- ------------------------------------------------------------
-- Computer identity
-- ------------------------------------------------------------
local function resolveComputerId()
  local id = nil
  if os.getComputerID then
    local cid = os.getComputerID()
    if cid and cid ~= 0 then id = tostring(cid) end
  end
  if not id and fs.exists(ID_PATH) then
    local f = fs.open(ID_PATH, "r")
    if f then
      id = trim(f.readAll())
      f.close()
    end
  end
  if not id or id == "" then
    id = randomHex(16)
    local f = fs.open(ID_PATH, "w")
    f.write(id)
    f.close()
  end
  return id
end

local computerId = resolveComputerId()

-- ------------------------------------------------------------
-- Mount peripheral discovery
-- ------------------------------------------------------------
local function findMount()
  if peripheral.find then
    local name, wrapped = peripheral.find(MOUNT_TYPE)
    if wrapped then return wrapped end
  end
  for _, side in ipairs(peripheral.getNames()) do
    if peripheral.getType(side) == MOUNT_TYPE then
      return peripheral.wrap(side)
    end
  end
  return nil
end

-- ------------------------------------------------------------
-- GPS location
-- ------------------------------------------------------------
-- GPS runs in its own worker coroutine (see the bottom of this file) so a
-- fresh fix is maintained continuously, even while the main loop is busy
-- polling the website or running the multi-second fire sequence.

-- Shared state: gpsWorker() updates this, the main loop reads it.
local gpsState = {
  x = nil, y = nil, z = nil,
  everFixed = false,
  reason = "Locating…",
}

-- Seconds to wait between GPS fix attempts. Combined with GPS_TIMEOUT this
-- keeps a new fix flowing roughly every 3 seconds (matching the website).
local GPS_POLL_INTERVAL = 1

local function findWirelessModem()
  for _, side in ipairs(peripheral.getNames()) do
    if peripheral.getType(side) == "modem" then
      local m = peripheral.wrap(side)
      if m.isWireless and m.isWireless() then
        return m
      end
    end
  end
  return nil
end

-- One GPS fix attempt. On success stores the fix in gpsState; on failure
-- records a human-readable reason for the status screen.
local function gpsTick()
  if not (gps and gps.locate) then
    gpsState.reason = "gps API unavailable"
    return
  end
  if not findWirelessModem() then
    gpsState.reason = "No wireless modem attached"
    return
  end
  local ok, x, y, z = pcall(gps.locate, GPS_TIMEOUT)
  if ok and x ~= nil and y ~= nil and z ~= nil then
    gpsState.x, gpsState.y, gpsState.z = x, y, z
    gpsState.everFixed = true
    gpsState.reason = nil
  else
    gpsState.reason = "No GPS fix — hosts out of range or not loaded"
  end
end

-- Concurrent GPS worker. Never throws, so one bad fix attempt can't take
-- down the whole daemon.
local function gpsWorker()
  while true do
    local ok, err = pcall(gpsTick)
    if not ok then
      gpsState.reason = "GPS error: " .. tostring(err)
    end
    sleep(GPS_POLL_INTERVAL)
  end
end

-- First-boot GPS self-check: prints the exact reason GPS is (or isn't)
-- working before the daemon loop takes over the screen. A successful fix is
-- reused as the daemon's starting position so it never flashes "no fix yet".
local function gpsSelfCheck()
  term.clear()
  term.setCursorPos(1, 1)
  print("=== GPS self-check ===")
  if not (gps and gps.locate) then
    print("FAIL: the gps API is unavailable on this computer.")
    return
  end
  if not findWirelessModem() then
    print("FAIL: no wireless modem attached.")
    print("       Attach a wireless (or ender) modem so GPS can work.")
    return
  end
  -- debug=true makes CC print its own diagnostics (the exact reason a fix
  -- could not be established, distances to each host, etc.).
  local ok, x, y, z = pcall(gps.locate, GPS_TIMEOUT, true)
  if not ok then
    print("FAIL: gps.locate errored: " .. tostring(x))
    return
  end
  if x and y and z then
    gpsState.x, gpsState.y, gpsState.z = x, y, z
    gpsState.everFixed = true
    gpsState.reason = nil
    print("OK: position is " .. tostring(x) .. ", " .. tostring(y) .. ", " .. tostring(z))
  else
    print("FAIL: gps.locate could not establish a position.")
    print("       GPS hosts must run 'gps host x y z', stay chunk-loaded,")
    print("       and be within wireless range of this computer.")
  end
end

-- ------------------------------------------------------------
-- Ship heading (two-point GPS)
-- ------------------------------------------------------------
-- Two dedicated computers run the "Sublevel Ship GPS" program: one at the
-- FRONT (bow) and one at the BACK (stern). Both broadcast their position on
-- the shared channel; this cannon computer listens to both and derives the
-- ship's heading as the direction from back to front, in the same yaw
-- convention as the website (0 = south, 90 = west, 180 = north, 270 = east).
local SHIP_GPS_CHANNEL = 4120   -- must match the Ship GPS program

local frontX, frontY, frontZ = nil, nil, nil
local backX,  backY,  backZ  = nil, nil, nil

local function computeShipYaw()
  if frontX == nil or backX == nil then return nil end
  local dx = frontX - backX
  local dz = frontZ - backZ
  if math.abs(dx) < 0.001 and math.abs(dz) < 0.001 then return nil end
  local deg = math.deg(math.atan2(-dx, dz))
  return ((deg % 360) + 360) % 360
end

-- Concurrent listener for the front/back GPS computers' broadcasts.
local function shipGpsListener()
  while true do
    local modem = findWirelessModem()
    if modem then
      modem.open(SHIP_GPS_CHANNEL)
      while true do
        local e, side, chan, replyChan, msg = os.pullEvent("modem_message")
        if e == "modem_message" and chan == SHIP_GPS_CHANNEL then
          local ok, data = pcall(textutils.unserializeJSON, msg)
          if ok and type(data) == "table" and data.type == "ship_gps" and data.x and data.z then
            if data.role == "back" then
              backX, backY, backZ = data.x, data.y, data.z
            else
              frontX, frontY, frontZ = data.x, data.y, data.z
            end
          end
        end
      end
    end
    sleep(5)
  end
end

-- ------------------------------------------------------------
-- HTTP bridge
-- ------------------------------------------------------------
local function httpPost(path, body)
  local url = BASE_URL .. path
  local headers = { ["Content-Type"] = "application/json" }
  local bodyStr = textutils.serializeJSON(body)
  local ok, result = pcall(function()
    local req = http.post(url, bodyStr, headers)
    if req then
      local raw = req.readAll()
      req.close()
      return raw
    end
    return nil
  end)
  if not ok or result == nil then return nil end
  local ok2, decoded = pcall(textutils.unserializeJSON, result)
  if ok2 and decoded then return decoded end
  return nil
end

-- ------------------------------------------------------------
-- Firing sequence (same as static cannons):
--   disassemble → assemble → aim → fire → disassemble
-- ------------------------------------------------------------
local function normDeg(d)
  d = d % 360
  if d > 180 then d = d - 360 end
  return d
end

local function waitAimed(mount, aimYaw, aimPitch)
  local start = os.clock()
  while os.clock() - start < AIM_TIMEOUT do
    local y, p
    pcall(function() y = mount.getYaw() end)
    pcall(function() p = mount.getPitch() end)
    if y ~= nil and p ~= nil then
      local dy = math.abs(normDeg(y - aimYaw))
      local dp = math.abs(p - aimPitch)
      if dy <= AIM_TOLERANCE and dp <= AIM_TOLERANCE then
        return true
      end
    end
    sleep(0.1)
  end
  return false
end

local function doFire(mount)
  if cfg.fireMode == "redstone" and cfg.redstoneSide ~= "" then
    redstone.setOutput(cfg.redstoneSide, true)
    sleep(cfg.firePulse)
    redstone.setOutput(cfg.redstoneSide, false)
    return true
  end
  return pcall(function() mount.fire() end)
end

local function runCommand(mount, cmd, shipYaw)
  print("Command #" .. tostring(cmd.sequence) .. ": aiming…")

  -- 1. Disassemble
  pcall(function() mount.disassemble() end)
  sleep(SEQ_DISASSEMBLE)

  -- 2. Assemble
  pcall(function() mount.assemble() end)
  sleep(SEQ_ASSEMBLE)

  -- 3. Delay before aiming
  sleep(SEQ_BEFORE_AIM)

  -- 4. Aim. The website sends an absolute world yaw; the mount's yaw is
  -- measured relative to the ship, so subtract the ship's heading first.
  local aimYaw   = (cmd.yaw or 0) + cfg.yawOffset
  if cfg.shipRelativeYaw and shipYaw then
    aimYaw = aimYaw - shipYaw
  end
  local aimPitch = (cmd.pitch or 0) * cfg.pitchScale
  pcall(function() mount.setYaw(aimYaw) end)
  pcall(function() mount.setPitch(aimPitch) end)
  local settled = waitAimed(mount, aimYaw, aimPitch)
  if not settled then
    print("Warning: aim did not settle — firing anyway")
  end

  -- 5. Delay before firing
  sleep(SEQ_BEFORE_FIRE)

  -- 6. Fire
  local fired = doFire(mount)
  if fired then
    print("Fired! (yaw " .. string.format("%.1f", aimYaw) .. ", pitch " .. string.format("%.1f", aimPitch) .. ")")
  else
    print("Fire failed — is the cannon loaded and assembled?")
  end
  sleep(SEQ_AFTER_FIRE)

  -- 7. Disassemble
  pcall(function() mount.disassemble() end)
end

-- ------------------------------------------------------------
-- Status screen
-- ------------------------------------------------------------
local function drawStatus(lines)
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setCursorPos(1, 1)
  term.setTextColor(colors.white)
  for _, line in ipairs(lines) do
    print(line)
  end
end

-- ------------------------------------------------------------
-- Main
-- ------------------------------------------------------------
loadConfig()
if not readConfig() then
  setup()
  loadConfig()
end

-- Print a clear GPS diagnostic before the daemon loop takes over the screen.
gpsSelfCheck()
sleep(1)

local lastSequence = 0

local function mainLoop()
  while true do
    local lines = {}
    local mount = findMount()

    local curYaw, curPitch = 0, 0
    if mount then
      pcall(function() curYaw = mount.getYaw() or 0 end)
      pcall(function() curPitch = mount.getPitch() or 0 end)
    end

    -- Read the latest fix from the concurrent GPS worker. If we have never
    -- obtained a fix, x/y/z stay nil so the website keeps its last known
    -- position instead of snapping the cannon's dot to 0,0,0.
    local gpsOk = gpsState.everFixed
    local shipYaw = computeShipYaw()

    local body = {
      computerId = computerId,
      x = gpsState.x, y = gpsState.y, z = gpsState.z,
      gpsOk = gpsOk,
      shipYaw = shipYaw,
      length = cfg.length,
      facing = cfg.facing,
      sublevel = true,
      message = cfg.message,
      yaw = curYaw,
      pitch = curPitch,
      sequence = lastSequence,
    }

    local resp = httpPost(POLL_PATH, body)

    table.insert(lines, "Sublevel Cannon Computer")
    table.insert(lines, "Computer ID: " .. computerId)
    if gpsState.x ~= nil then
      local stale = gpsState.reason ~= nil
      table.insert(lines, "GPS: X " .. tostring(gpsState.x) .. "  Y " .. tostring(gpsState.y) .. "  Z " .. tostring(gpsState.z) .. (stale and " (stale)" or ""))
    else
      table.insert(lines, "GPS: no fix yet")
    end
    if gpsState.reason then
      table.insert(lines, "GPS issue: " .. gpsState.reason)
    end
    if shipYaw then
      table.insert(lines, "Ship heading: " .. string.format("%.1f", shipYaw) .. "°")
    elseif frontX == nil or backX == nil then
      table.insert(lines, "Ship heading: waiting for front/back GPS…")
    end

    if not mount then
      table.insert(lines, "Mount: not found")
      table.insert(lines, "Place this computer touching the cannon mount.")
    end

    if resp == nil then
      table.insert(lines, "Status: cannot reach website")
      table.insert(lines, "Check " .. BASE_URL)
    elseif not resp.success then
      table.insert(lines, "Status: server error")
      table.insert(lines, tostring(resp.response))
    elseif resp.response.status == "pending" then
      table.insert(lines, "Status: AWAITING APPROVAL")
      table.insert(lines, "An officer must accept this cannon")
      table.insert(lines, "from the website's Cannon Registry tab.")
    elseif resp.response.status == "active" then
      local cmd = resp.response.command
      if cmd and cmd.sequence ~= lastSequence then
        if mount then
          lastSequence = cmd.sequence   -- ack intent immediately (at-most-once fire)
          local ok, err = pcall(runCommand, mount, cmd, shipYaw)
          if not ok then
            print("Command #" .. tostring(cmd.sequence) .. " failed: " .. tostring(err))
          end
        else
          table.insert(lines, "Fire command queued — mount not found")
        end
      else
        table.insert(lines, "Status: READY")
        table.insert(lines, "Name: " .. tostring(resp.response.name))
        table.insert(lines, "Aiming: yaw " .. string.format("%.1f", curYaw) .. " / pitch " .. string.format("%.1f", curPitch))
      end
    else
      table.insert(lines, "Status: " .. tostring(resp.response.status))
    end

    drawStatus(lines)
    sleep(cfg.pollInterval)
  end
end

-- Run the GPS worker, the ship-heading listener, and the poll/fire loop at
-- the same time. Each gets its own coroutine; gps.locate(), the modem
-- listener and http.post() all yield, so none blocks the others.
parallel.waitForAll(gpsWorker, shipGpsListener, mainLoop)
