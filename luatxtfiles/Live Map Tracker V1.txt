local BASE_URL = "https://regnumaeternum.starpanda24825.workers.dev"
local MAPVIEW_PATH = "/api/mapview"
local BLUEMAP_URL = "http://regnumaeternum.enderman.cloud:50500"
local OVERWORLD = "world"
local CFG_PATH = "mapview_cfg.txt"
local BLOCK = string.char(223)
local ZOOMS = { 1, 2, 3, 4, 6 }
local COLOR_BIT = { 1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384, 32768 }
local HEADER_ROWS = 1
local FOOTER_ROWS = 1

local KEY_ENTER = (keys and keys.enter) or 28
local KEY_NUMPAD_ENTER = (keys and keys.numPadEnter) or 156
local KEY_BACKSPACE = (keys and keys.backspace) or 14
local KEY_UP = (keys and keys.up) or 200
local KEY_DOWN = (keys and keys.down) or 208
local KEY_LEFT = (keys and keys.left) or 203
local KEY_RIGHT = (keys and keys.right) or 205

local DIGIT_CODES = {
  [2] = "1", [3] = "2", [4] = "3", [5] = "4", [6] = "5", [7] = "6", [8] = "7", [9] = "8", [10] = "9", [11] = "0",
  [79] = "1", [80] = "2", [81] = "3", [75] = "4", [76] = "5", [77] = "6", [71] = "7", [72] = "8", [73] = "9", [82] = "0",
}

local cfg = { player = "", zoom = 2, refresh = 5 }

local mon
local monSide
local lastPalette
local lastPx
local pickerIndex = 1

local function isEnter(code)
  if code == KEY_ENTER then return true end
  if KEY_NUMPAD_ENTER and code == KEY_NUMPAD_ENTER then return true end
  return false
end

local function pad(text, width)
  text = tostring(text or "")
  if width < 1 then return "" end
  if #text >= width then return text:sub(1, width) end
  return text .. string.rep(" ", width - #text)
end

local function twoColumn(width, left, right)
  left = tostring(left or "")
  right = tostring(right or "")
  if #right > width then right = right:sub(1, width) end
  if #left + #right + 1 > width then
    left = left:sub(1, math.max(0, width - #right - 1))
  end
  local gap = width - #left - #right
  if gap < 0 then gap = 0 end
  return left .. string.rep(" ", gap) .. right
end

local function round(value)
  return math.floor((tonumber(value) or 0) + 0.5)
end

local function getJSON(url)
  local ok, res = pcall(http.get, url)
  if not ok or not res then return nil, 0 end
  local code = 200
  if res.getResponseCode then
    local okCode, value = pcall(res.getResponseCode)
    if okCode and tonumber(value) then code = tonumber(value) end
  end
  local body = res.readAll()
  res.close()
  local okBody, data = pcall(textutils.unserializeJSON, body)
  if okBody and type(data) == "table" then return data, code end
  return nil, code
end

local function loadConfig()
  if not fs.exists(CFG_PATH) then return end
  local handle = fs.open(CFG_PATH, "r")
  if not handle then return end
  local raw = handle.readAll()
  handle.close()
  local ok, data = pcall(textutils.unserializeJSON, raw)
  if not ok or type(data) ~= "table" then return end
  if type(data.player) == "string" then cfg.player = data.player end
  if tonumber(data.zoom) then cfg.zoom = math.floor(tonumber(data.zoom)) end
  if tonumber(data.refresh) then cfg.refresh = math.max(2, math.floor(tonumber(data.refresh))) end
end

local function saveConfig()
  local handle = fs.open(CFG_PATH, "w")
  if not handle then return end
  handle.write(textutils.serializeJSON({ player = cfg.player, zoom = cfg.zoom, refresh = cfg.refresh }))
  handle.close()
end

local function findMonitor()
  for _, side in ipairs(peripheral.getNames()) do
    if peripheral.getType(side) == "monitor" then
      local found = peripheral.wrap(side)
      if found then
        monSide = side
        mon = found
        pcall(mon.setTextScale, 0.5)
        lastPalette = nil
        lastPx = nil
        return true
      end
    end
  end
  monSide = nil
  mon = nil
  return false
end

local function fetchPlayers()
  local players = {}
  local data, code = getJSON(BLUEMAP_URL .. "/maps/" .. OVERWORLD .. "/live/players.json")
  if not data then return players, code > 0 end
  if type(data.players) == "table" then
    for _, entry in ipairs(data.players) do
      if entry.foreign ~= true and entry.name then
        local position = entry.position or {}
        local rotation = entry.rotation or {}
        table.insert(players, {
          name = tostring(entry.name),
          x = tonumber(position.x) or 0,
          y = tonumber(position.y) or 0,
          z = tonumber(position.z) or 0,
          yaw = tonumber(rotation.yaw) or 0,
        })
      end
    end
  end
  table.sort(players, function(a, b) return a.name < b.name end)
  return players, true
end

local function fetchView(x, z, w, h, zoom, yaw)
  local url = string.format(
    "%s%s?map=%s&x=%.2f&z=%.2f&w=%d&h=%d&zoom=%d&yaw=%.0f",
    BASE_URL, MAPVIEW_PATH, OVERWORLD, x, z, math.floor(w), math.floor(h), math.floor(zoom), yaw or 0
  )
  local data = getJSON(url)
  if data and data.success and type(data.response) == "table" and type(data.response.px) == "string" then
    return data.response
  end
  return nil
end

local function applyPalette(hex)
  if not mon or type(hex) ~= "string" or #hex < 96 or hex == lastPalette then return end
  for index = 0, 15 do
    local rgb = tonumber(hex:sub(index * 6 + 1, index * 6 + 6), 16)
    if rgb then pcall(mon.setPaletteColour, COLOR_BIT[index + 1], rgb) end
  end
  lastPalette = hex
end

local function drawMap(view)
  if not mon then return end
  local w = math.floor(tonumber(view.w) or 0)
  local h = math.floor(tonumber(view.h) or 0)
  local px = view.px
  if w < 1 or h < 2 or type(px) ~= "string" then return end
  if #px < w * h then px = px .. string.rep("f", w * h - #px) end
  if px == lastPx then return end

  local mw, mh = mon.getSize()
  local rows = math.floor(h / 2)
  local room = mh - HEADER_ROWS - FOOTER_ROWS
  if rows > room then rows = room end

  for row = 0, rows - 1 do
    local text, fg, bg = {}, {}, {}
    local base = row * 2 * w
    for column = 1, mw do
      if column <= w then
        text[column] = BLOCK
        local top = px:sub(base + column, base + column)
        local bottom = px:sub(base + w + column, base + w + column)
        fg[column] = top ~= "" and top or "f"
        bg[column] = bottom ~= "" and bottom or "f"
      else
        text[column] = " "
        fg[column] = "f"
        bg[column] = "f"
      end
    end
    mon.setCursorPos(1, HEADER_ROWS + 1 + row)
    mon.blit(table.concat(text), table.concat(fg), table.concat(bg))
  end
  lastPx = px
end

local function drawMonitorFrame(headerLeft, headerRight, footerLeft, footerRight)
  if not mon then return end
  local mw, mh = mon.getSize()
  if mh < 3 or mw < 4 then return end
  mon.setBackgroundColor(colors.black)
  mon.setTextColor(colors.white)
  mon.setCursorPos(1, 1)
  mon.write(twoColumn(mw, " " .. headerLeft, headerRight .. " "))
  mon.setCursorPos(1, mh)
  mon.write(twoColumn(mw, " " .. footerLeft, footerRight .. " "))
end

local function drawIdleMonitor(players, note)
  if not mon then return end
  local ok = pcall(function()
    local mw, mh = mon.getSize()
    mon.setBackgroundColor(colors.black)
    mon.setTextColor(colors.white)
    mon.clear()
    mon.setCursorPos(1, 1)
    mon.write(pad("LIVE MAP - Regnum Aeternum", mw))
    mon.setCursorPos(1, 2)
    mon.write(pad(note, mw))
    local line = 4
    for index, player in ipairs(players) do
      if line > mh then break end
      mon.setCursorPos(1, line)
      mon.write(pad(string.format("%-3d%-14s X %5d Y %3d Z %5d", index, player.name, round(player.x), round(player.y), round(player.z)), mw))
      line = line + 1
    end
  end)
  if not ok then findMonitor() end
end

local function putLine(x, y, text, colour)
  local tw = term.getSize()
  if y < 1 then y = 1 end
  term.setCursorPos(x, y)
  term.setTextColor(colour or colors.white)
  term.write(pad(text, tw - x + 1))
end

local function drawConsole(name, x, y, z, logged, count, reachable, view)
  local _, th = term.getSize()
  term.setBackgroundColor(colors.black)
  putLine(1, 1, "LIVE MAP TRACKER", colors.white)
  putLine(1, 3, "Tracking: " .. name, colors.lime)
  putLine(1, 4, "World:    " .. OVERWORLD, colors.lightGray)
  if x then
    putLine(1, 5, string.format("Position: X %d  Y %d  Z %d", round(x), round(y), round(z)), colors.lightGray)
  else
    putLine(1, 5, "Position: waiting for a live fix", colors.yellow)
  end
  putLine(1, 6, string.format("Players:  %d online", count), colors.lightGray)
  if view then
    putLine(1, 7, string.format("Zoom:     %d blocks/px", view.bx or 0), colors.lightGray)
    putLine(1, 8, string.format("Frame:    %dx%d pixels", view.w or 0, view.h or 0), colors.lightGray)
  end
  if not reachable then
    putLine(1, 10, "BlueMap unreachable from this computer.", colors.red)
  elseif not logged then
    putLine(1, 10, name .. " is not in the overworld - holding the last known position.", colors.yellow)
  end
  local monOk, mw, mh
  if mon then
    monOk, mw, mh = pcall(mon.getSize)
  end
  if monOk then
    putLine(1, th - 4, string.format("Monitor: %s  %dx%d  %d blocks per pixel", monSide, mw, mh, view and view.bx or 0), colors.gray)
  else
    putLine(1, th - 4, "Monitor: none attached", colors.red)
  end
  putLine(1, th - 3, "[z] zoom   [p] or [enter] player list   [touch monitor] next player", colors.gray)
  putLine(1, th, "", colors.gray)
  term.setBackgroundColor(colors.black)
  term.setCursorPos(1, th)
end

local function pickStatus(players, reachable, message)
  if message ~= "" then return message end
  if not reachable then
    return "Cannot reach " .. BLUEMAP_URL .. " from this computer."
  end
  if #players == 0 then
    return "No players are in the overworld right now."
  end
  return string.format("%d player(s) online   zoom %d = %d blocks per pixel", #players, cfg.zoom, cfg.zoom * 4)
end

local function drawPicker(players, selected, typed, offset, status)
  local tw, th = term.getSize()
  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  term.clear()
  term.setCursorPos(1, 1)
  term.write("LIVE MAP TRACKER")
  term.setTextColor(colors.gray)
  term.setCursorPos(1, 2)
  term.write("Live overworld map on the monitor")
  term.setTextColor(colors.white)
  term.setCursorPos(1, 3)
  term.write(string.format("%-4s%-16s%6s%5s%6s", "#", "PLAYER", "X", "Y", "Z"))

  local room = th - 6
  if room < 1 then room = 1 end
  if selected > offset + room then offset = selected - room end
  if selected <= offset then offset = selected - 1 end
  if offset < 1 then offset = 1 end

  for row = 1, room do
    local index = offset + row
    local player = players[index]
    term.setCursorPos(1, 3 + row)
    if player then
      local text = string.format("%-4d%-16s%6d%5d%6d", index, player.name, round(player.x), round(player.y), round(player.z))
      if index == selected then
        term.setBackgroundColor(colors.gray)
        term.setTextColor(colors.black)
      else
        term.setBackgroundColor(colors.black)
        term.setTextColor(colors.white)
      end
      term.write(pad(text, tw))
    else
      term.setBackgroundColor(colors.black)
      term.setTextColor(colors.white)
      term.write(string.rep(" ", tw))
    end
  end

  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  putLine(1, th - 2, "Number, arrows + enter   r refresh   q quit", colors.lightGray)
  putLine(1, th - 1, "> " .. typed, colors.yellow)
  putLine(1, th, status, colors.gray)
  return offset
end

local function pickPlayer(players, reachable)
  local list = players
  local selected = pickerIndex
  if selected < 1 or selected > #list then selected = 1 end
  local typed = ""
  local message = ""
  local offset = 1

  local function clamp()
    if selected > #list then selected = #list end
    if selected < 1 then selected = 1 end
    pickerIndex = selected
  end

  local function draw()
    clamp()
    offset = drawPicker(list, selected, typed, offset, pickStatus(list, reachable, message))
  end

  local function reload()
    local fresh, ok = fetchPlayers()
    list = fresh
    reachable = ok
    message = ""
    draw()
  end

  local function commit()
    local index = tonumber(typed)
    if index and list[index] then
      pickerIndex = index
      return list[index]
    end
    if typed ~= "" and #list > 0 then
      typed = ""
      message = "Type a number between 1 and " .. tostring(#list) .. "."
      draw()
      return nil
    end
    if list[selected] then
      pickerIndex = selected
      return list[selected]
    end
    message = pickStatus(list, reachable, "")
    draw()
    return nil
  end

  local typedFrom = selected

  local function typeDigit(digit)
    message = ""
    if #typed == 0 then typedFrom = selected end
    if #typed < 3 then
      typed = typed .. digit
      local index = tonumber(typed)
      if index and index >= 1 and index <= #list then
        selected = index
      end
    end
    draw()
  end

  local function backspace()
    typed = typed:sub(1, math.max(0, #typed - 1))
    if typed == "" then
      selected = typedFrom
    else
      local index = tonumber(typed)
      if index and list[index] then selected = index end
    end
    draw()
  end

  draw()
  local timer = os.startTimer(cfg.refresh)
  local heldDigit
  local heldTimer

  local function clearHeld()
    if heldTimer then os.cancelTimer(heldTimer) end
    heldDigit = nil
    heldTimer = nil
  end

  while true do
    local event, a = os.pullEvent()

    if event == "timer" then
      if a == timer then
        drawPicker(list, selected, typed, offset, "Refreshing the player list...")
        reload()
        timer = os.startTimer(cfg.refresh)
      elseif a == heldTimer then
        local digit = heldDigit
        clearHeld()
        if digit then typeDigit(digit) end
      end

    elseif event == "key" then
      if isEnter(a) then
        local choice = commit()
        if choice then return choice end
      elseif a == KEY_UP or a == KEY_LEFT then
        message = ""
        selected = selected - 1
        typed = ""
        typedFrom = selected
        draw()
      elseif a == KEY_DOWN or a == KEY_RIGHT then
        message = ""
        selected = selected + 1
        typed = ""
        typedFrom = selected
        draw()
      elseif a == KEY_BACKSPACE then
        message = ""
        backspace()
      elseif DIGIT_CODES[a] then
        clearHeld()
        heldDigit = DIGIT_CODES[a]
        heldTimer = os.startTimer(0.05)
      end

    elseif event == "char" then
      local key = string.lower(a)
      if key:match("%d") then
        if heldDigit == key then clearHeld() end
        typeDigit(key)
      elseif key == "\n" or key == "\r" then
        local choice = commit()
        if choice then return choice end
      elseif key == "q" then
        return nil
      elseif key == "r" then
        reload()
        timer = os.startTimer(cfg.refresh)
      end

    elseif event == "monitor_touch" then
      if a == monSide and #list > 0 then
        message = ""
        selected = selected % #list + 1
        typed = ""
        typedFrom = selected
        draw()
      end

    elseif event == "monitor_resize" or event == "peripheral" or event == "peripheral_detach" then
      findMonitor()
      draw()
    end
  end
end

local function nextPlayerName(players, current)
  if #players == 0 then return nil end
  local index
  for i, player in ipairs(players) do
    if player.name == current then index = i end
  end
  if not index then return players[1].name end
  return players[index % #players + 1].name
end

local function footerText(x, z, logged, reachable, count, view)
  local parts = {}
  if x then
    parts[#parts + 1] = string.format("X %d Z %d", round(x), round(z))
  else
    parts[#parts + 1] = "no live fix"
  end
  if view and view.zoom then
    parts[#parts + 1] = "z" .. tostring(view.zoom)
  end
  parts[#parts + 1] = tostring(count) .. " online"
  if not reachable then
    parts[#parts + 1] = "bluemap down"
  elseif not logged then
    parts[#parts + 1] = "not in overworld"
  end
  return table.concat(parts, "  ")
end

local function track(name)
  local x, y, z, yaw
  local logged = false
  local reachable = true
  local view
  local waiting = false
  local players = {}
  local zoomIndex = 1
  for i, value in ipairs(ZOOMS) do
    if value == cfg.zoom then zoomIndex = i end
  end

  lastPx = nil
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setCursorPos(1, 1)

  while true do
    local fresh, ok = fetchPlayers()
    players = fresh
    reachable = ok

    local me
    for _, player in ipairs(players) do
      if player.name == name then me = player end
    end
    if me then
      x, y, z, yaw, logged = me.x, me.y, me.z, me.yaw, true
    else
      logged = false
    end

    if mon then
      local sized, mw, mh = pcall(mon.getSize)
      if sized and x and mw > 1 and mh > HEADER_ROWS + FOOTER_ROWS + 1 then
        local drawn = fetchView(x, z, mw, 2 * (mh - HEADER_ROWS - FOOTER_ROWS), ZOOMS[zoomIndex], yaw)
        if drawn then view = drawn end
      end
      local drawnOk = pcall(function()
        local frameWidth, frameHeight = mon.getSize()
        if view then
          applyPalette(view.palette)
          drawMap(view)
          waiting = false
        elseif not waiting then
          waiting = true
          mon.setBackgroundColor(colors.black)
          mon.setTextColor(colors.white)
          mon.clear()
          mon.setCursorPos(1, math.max(1, math.floor(frameHeight / 2)))
          mon.write(pad(" waiting for the map service", frameWidth))
        end
        drawMonitorFrame(name, OVERWORLD, footerText(x, z, logged, reachable, #players, view), os.date("%H:%M"))
      end)
      if not drawnOk then findMonitor() end
    end

    drawConsole(name, x, y, z, logged, #players, reachable, view)

    local timer = os.startTimer(cfg.refresh)
    local redraw = false
    while not redraw do
      local event, a = os.pullEvent()
      if event == "timer" and a == timer then
        redraw = true
      elseif event == "key" then
        if isEnter(a) then
          return
        end
      elseif event == "char" then
        local key = string.lower(a)
        if key == "z" then
          zoomIndex = zoomIndex % #ZOOMS + 1
          cfg.zoom = ZOOMS[zoomIndex]
          saveConfig()
          lastPx = nil
          view = nil
          waiting = false
          redraw = true
        elseif key == "p" then
          return
        end
      elseif event == "monitor_touch" and a == monSide then
        local nextName = nextPlayerName(players, name)
        if nextName and nextName ~= name then
          name = nextName
          cfg.player = name
          saveConfig()
          x, y, z, yaw = nil, nil, nil, nil
          view = nil
          waiting = false
          lastPx = nil
          redraw = true
        end
      elseif event == "monitor_resize" or event == "peripheral" or event == "peripheral_detach" then
        findMonitor()
        redraw = true
      end
    end
  end
end

local function main()
  loadConfig()

  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  term.clear()
  term.setCursorPos(1, 1)

  if not findMonitor() then
    print("LIVE MAP TRACKER")
    print("")
    print("Attach a monitor to this computer.")
    print("Waiting for a monitor...")
    sleep(2)
    while not findMonitor() do
      sleep(2)
    end
    print("")
    print("Monitor found on the " .. tostring(monSide) .. " side.")
    sleep(1)
  end

  if mon and mon.isColour and not mon.isColour() then
    print("")
    print("This is a basic monitor - the map will be shown in greyscale.")
    sleep(2)
  end

  while true do
    local players, reachable = fetchPlayers()
    drawIdleMonitor(players, "Pick a player on the computer")
    local choice = pickPlayer(players, reachable)
    if choice then
      cfg.player = choice.name
      saveConfig()
      track(choice.name)
    else
      if mon then
        pcall(function()
          local mw = mon.getSize()
          mon.setBackgroundColor(colors.black)
          mon.clear()
          mon.setTextColor(colors.white)
          mon.setCursorPos(1, 1)
          mon.write(pad("LIVE MAP - idle", mw))
        end)
      end
      term.setBackgroundColor(colors.black)
      term.setTextColor(colors.white)
      term.clear()
      term.setCursorPos(1, 1)
      print("Stopped.")
      return
    end
  end
end

main()
