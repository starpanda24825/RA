-- ============================================================
-- Sublevel Ship GPS — Regnum Aeternum
-- ComputerCraft FRONT / BACK position beacon for mobile/sublevel ships.
-- ============================================================
--
-- WHAT THIS IS
--   A sublevel ship needs two GPS points to know which way it is pointing.
--   This program runs on TWO separate computers on the ship: one at the FRONT
--   (bow) and one at the BACK (stern), placed as far apart as possible along
--   the fore-aft centerline. Each broadcasts its position over the wireless
--   modem. A third computer listens to both and derives the ship's heading:
--   either each Sublevel Cannon Computer (for guns running standalone) or, on
--   a ship with a Sublevel Vehicle Computer, that vehicle — which then aims
--   every cannon from one shared heading.
--
-- WHAT CHANGED FROM THE OLD VERSION
--   It now uses the gpsnet engine (GPS Network) instead of gps.locate().
--   Three things follow from that:
--
--   1) IT NO LONGER STALLS. gps.locate() blocks for up to its whole timeout,
--      and the old loop sat in it for two thirds of every cycle. Here a modem
--      coroutine does all the listening and the beacon reads a cached fix, so
--      the screen stays live and the broadcast cadence stays steady.
--
--   2) SIMULTANEOUS SAMPLING. The heading is the direction from the stern fix
--      to the bow fix, so if the ship is TURNING while the two beacons take
--      their fixes at different moments, the two points do not describe one
--      rigid ship and the heading is wrong — by roughly half the turn rate
--      times the gap between fixes. A ship turning at 10°/s with fixes a
--      second apart would be 5° out, which is a lot at long range.
--      The cannon computer now broadcasts a SYNC tick on channel 4120 and
--      both beacons take a fresh fix the instant they hear it, so their two
--      positions come from the same instant. Each broadcast carries the age
--      of the fix it used, so the cannon computer can see the remaining skew.
--      If no SYNC ever arrives the beacon just free-runs as before.
--
--   3) FIXES ARE VALIDATED. Each fix is solved from the whole tower set and
--      checked against its own residuals, so a tower that has been moved or
--      mis-keyed is detected and left out instead of bending the heading.
--
-- ============================================================
-- IN-GAME SETUP (run this on BOTH computers)
-- ============================================================
--   1. Place a Computer at the bow and another at the stern, on the
--      centerline, as far apart as the ship allows.
--   2. Put a wireless (or ender) modem on each, and give the computer a
--      wireless network connection (http must work).
--   3. Install this program as "startup", and install the GPS Network
--      library next to it as "gpsnet.lua":
--
--        wget https://raw.githubusercontent.com/starpanda24825/RA/refs/heads/main/luatxtfiles/Cannon%20Programs/GPS%20Network%20V1.txt gpsnet.lua
--
--      Both files must sit in the same folder. You also need FOUR OR MORE
--      towers running the "GPS Tower" program (see that file) within wireless
--      range of the ship.
--   4. On first boot each asks whether it is the FRONT or BACK computer and
--      saves that role. Answer FRONT for the bow computer and BACK for the
--      stern one.
--   5. It then shows the live registry pulled from the website and asks
--      which ship this beacon pair belongs to. Use the UP / DOWN arrow keys
--      to highlight and ENTER to confirm. Both computers (front and back)
--      must pick the SAME entry. The list re-fetches itself every 10 seconds.
--
-- ASSIGNMENT — WHY IT MATTERS, AND WHICH KIND TO PICK
--   Every broadcast is tagged with the chosen entry's registry id, and only
--   that entry's computers accept it. Two ships parked near each other
--   therefore cannot swap front/back fixes, so a shipYaw can never be applied
--   to the wrong vehicle.
--
--   The list offers both kinds of entry, because which is correct depends on
--   how the ship is actually set up on the website:
--
--     * VEHICLE — pick this when the ship's cannons are assigned to a
--       Sublevel Vehicle Computer. The beacons then name the SHIP itself,
--       which is the honest description of what a bow/stern pair measures, and
--       the vehicle accepts them for every gun it controls.
--     * CANNON — pick this for a cannon that has NOT been assigned to a
--       vehicle. A standalone cannon computer derives the heading for itself,
--       so it needs beacons addressed to it by name.
--
--   In doubt: if the cannon you would otherwise have picked is shown as
--   "[on Vehicle N]", assign the beacons to that vehicle instead.
--
--   Press C at any time on the status screen to re-open the list and change
--   the assignment.
--
-- HOW FAR APART SHOULD THE TWO BEACONS BE?
--   As far as the ship allows — it is the single biggest factor in aiming
--   accuracy. Heading error is roughly 57.3 x (fix error / separation), so
--   with fixes good to a block, a 20-block separation is ~3° of heading error
--   and a 60-block separation is ~1°. Squeezing more precision out of GPS
--   below one block buys nothing; moving these two computers apart does.
-- ============================================================

-- ------------------------------------------------------------
-- Configuration
-- ------------------------------------------------------------
local BASE_URL      = "https://regnumaeternum.starpanda24825.workers.dev"
local CANNONS_PATH  = "/api/ballistics/cc/cannons"
local VEHICLES_PATH = "/api/ballistics/cc/vehicles"

-- Channel the positions are broadcast on, and the channel the cannon
-- computer sends its SYNC ticks on. MUST match SHIP_GPS_CHANNEL in the
-- Sublevel Cannon Computer program.
local CHANNEL = 4120

local REPORT_INTERVAL = 1    -- fallback seconds between broadcasts when no SYNC
local SYNC_WAIT       = 1    -- seconds to wait for a fix after a SYNC tick
local FIX_FRESH_AGE   = 0.6  -- a fix younger than this counts as "just taken"
local REFRESH_SECONDS = 10   -- how often the cannon list re-fetches itself
local CFG_PATH        = "shipgps_cfg.txt"

-- ------------------------------------------------------------
-- Helpers
-- ------------------------------------------------------------
local function trim(s)
  s = tostring(s or "")
  s = s:gsub("^%s+", "")
  s = s:gsub("%s+$", "")
  return s
end

-- Blocking GET returning the decoded JSON table (or nil on any failure).
local function httpGetJSON(path)
  if not http then return nil end
  local ok, resp = pcall(http.get, BASE_URL .. path)
  if not ok or not resp then return nil end
  local raw = resp.readAll()
  resp.close()
  local decoded, data = pcall(textutils.unserializeJSON, raw)
  if decoded and type(data) == "table" then return data end
  return nil
end

-- ------------------------------------------------------------
-- Screen
-- ------------------------------------------------------------
-- Only lines whose contents actually changed are redrawn. Terminal writes are
-- among the most expensive things a CC computer does, and this program is a
-- daemon: repainting an identical screen several times a second is exactly
-- the kind of self-inflicted load that makes a network feel slow.
local screenCache = {}

local function resetScreen()
  screenCache = {}
  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.white)
  term.clear()
end

local function drawScreen(lines)
  local width = term.getSize()
  local count = #lines
  if #screenCache > count then count = #screenCache end
  for i = 1, count do
    local entry = lines[i]
    local text, colour = "", colors.white
    if type(entry) == "table" then
      text, colour = entry[1] or "", entry[2] or colors.white
    elseif type(entry) == "string" then
      text = entry
    end
    if #text > width then text = text:sub(1, width) end
    local padded = text .. string.rep(" ", math.max(0, width - #text))
    local key = tostring(colour) .. "\1" .. padded
    if screenCache[i] ~= key then
      term.setBackgroundColor(colors.black)
      term.setTextColor(colour)
      term.setCursorPos(1, i)
      term.write(padded)
      screenCache[i] = key
    end
  end
end

-- ------------------------------------------------------------
-- Config  (role + assigned cannon)
-- ------------------------------------------------------------
local cfg = {
  role        = nil,   -- "front" | "back"
  -- Exactly ONE of the two pairs below is set: the registry entry this beacon
  -- pair reports for. A vehicle is normally right on a ship whose cannons
  -- belong to a Sublevel Vehicle Computer; a cannon is for a standalone gun.
  vehicleId   = nil,   -- vehicle registry id on the website
  vehicleName = nil,   -- display name, cached for the status screen
  cannonId    = nil,   -- cannon registry id on the website
  cannonName  = nil,
}

local function readConfig()
  if not fs.exists(CFG_PATH) then return end
  local f = fs.open(CFG_PATH, "r")
  if not f then return end
  local raw = trim(f.readAll())
  f.close()
  if raw == "" then return end

  -- Backwards compatibility: the original file was just the role on one line.
  if raw == "front" or raw == "back" then
    cfg.role = raw
    return
  end

  for line in raw:gmatch("[^\r\n]+") do
    local k, v = line:match("^([%w_]+)%s*=%s*(.-)%s*$")
    if k == "role" then
      if v == "front" or v == "back" then cfg.role = v end
    elseif k == "cannonId" then
      cfg.cannonId = tonumber(v)
    elseif k == "cannonName" then
      cfg.cannonName = v
    elseif k == "vehicleId" then
      cfg.vehicleId = tonumber(v)
    elseif k == "vehicleName" then
      cfg.vehicleName = v
    end
  end
end

local function writeConfig()
  local f = fs.open(CFG_PATH, "w")
  f.writeLine("role=" .. tostring(cfg.role or ""))
  f.writeLine("vehicleId=" .. tostring(cfg.vehicleId or ""))
  f.writeLine("vehicleName=" .. tostring(cfg.vehicleName or ""))
  f.writeLine("cannonId=" .. tostring(cfg.cannonId or ""))
  f.writeLine("cannonName=" .. tostring(cfg.cannonName or ""))
  f.close()
end

-- What this beacon pair is assigned to. Exactly one kind is ever set, so these
-- read as a single choice rather than a set of registrations.
local function assignmentName()
  if cfg.vehicleId then return cfg.vehicleName or ("Vehicle " .. tostring(cfg.vehicleId)) end
  if cfg.cannonId then return cfg.cannonName or ("Cannon " .. tostring(cfg.cannonId)) end
  return nil
end

local function assignmentKind()
  if cfg.vehicleId then return "vehicle" end
  if cfg.cannonId then return "cannon" end
  return nil
end

local function assignmentId()
  if cfg.vehicleId then return cfg.vehicleId end
  return cfg.cannonId
end

-- "Vehicle 1 (vehicle id 3)", for the confirmation screen.
local function assignmentLabel()
  local name = assignmentName()
  if not name then return "nothing" end
  return name .. " (" .. tostring(assignmentKind()) .. " id " .. tostring(assignmentId()) .. ")"
end

-- ------------------------------------------------------------
-- Role (front / back)
-- ------------------------------------------------------------
local function setupRole()
  resetScreen()
  term.setCursorPos(1, 1)
  print("=== Sublevel Ship GPS — first-time setup ===")
  print("")
  print("Which end of the ship is this computer on?")
  print("  1 = FRONT (bow)   — the ship's forward end")
  print("  2 = BACK  (stern) — the ship's rear")
  print("")
  write("Enter 1 or 2: ")
  local choice = trim(read())
  cfg.role = (choice == "2") and "back" or "front"
  writeConfig()
  print("")
  print("Saved as " .. cfg.role .. ".")
  sleep(1)
  resetScreen()
end

-- ------------------------------------------------------------
-- Registry selection (vehicles and cannons)
-- ------------------------------------------------------------
-- A beacon pair identifies the ship it sits on, and a ship can be owned by
-- either kind of registry entry:
--
--   * VEHICLE — a Sublevel Vehicle Computer coordinating every cannon on the
--     ship. This is the right choice whenever the ship's cannons are assigned
--     to one, and it is listed first for that reason: the beacons then name
--     the SHIP rather than one arbitrary gun standing in for it.
--   * CANNON — for a gun that has NOT been assigned to a vehicle. Those run
--     standalone and need beacons addressed to themselves.
--
-- A cannon that belongs to a vehicle is still listed, tagged with that
-- vehicle, so the questionable choice is obvious on the screen.
local function fetchVehicles()
  local data = httpGetJSON(VEHICLES_PATH)
  if data and data.success and data.response and type(data.response.vehicles) == "table" then
    local list = {}
    for _, v in ipairs(data.response.vehicles) do
      if type(v) == "table" and tonumber(v.id) then
        table.insert(list, v)
      end
    end
    table.sort(list, function(a, b) return tonumber(a.id) < tonumber(b.id) end)
    return list
  end
  return nil
end

local function fetchCannons()
  local data = httpGetJSON(CANNONS_PATH)
  if data and data.success and data.response and type(data.response.cannons) == "table" then
    local list = {}
    for _, c in ipairs(data.response.cannons) do
      if type(c) == "table" and tonumber(c.id) then
        table.insert(list, c)
      end
    end
    table.sort(list, function(a, b) return tonumber(a.id) < tonumber(b.id) end)
    return list
  end
  return nil
end

-- One list: vehicles first, then cannons. Returns nil only when NEITHER
-- registry could be read, so an operator is never offered a half-read list to
-- assign against.
local function fetchRegistry()
  local out = {}
  local vehicles = fetchVehicles()
  local vehicleNames = {}

  if vehicles then
    for _, v in ipairs(vehicles) do
      local id = tonumber(v.id)
      local name = v.name
      if not name or name == "" then name = "Vehicle " .. tostring(id) end
      vehicleNames[id] = name
      out[#out + 1] = {
        kind = "vehicle", id = id, name = name,
        computerId = v.computerId, status = v.status,
      }
    end
  end

  local cannons = fetchCannons()
  if cannons then
    for _, c in ipairs(cannons) do
      local vid = tonumber(c.vehicleId)
      out[#out + 1] = {
        kind = "cannon", id = tonumber(c.id), name = c.name,
        computerId = c.computerId, status = c.status, sublevel = c.sublevel,
        vehicleId = vid, vehicleName = vid and vehicleNames[vid] or nil,
      }
    end
  end

  if not vehicles and not cannons then return nil end
  return out
end

local function entryLabel(e)
  local name = e.name
  if not name or name == "" then
    name = (e.kind == "vehicle" and "Vehicle " or "Cannon ") .. tostring(e.id)
  end
  local tag = " [" .. e.kind .. "]"
  if e.kind == "cannon" then
    if e.sublevel then tag = tag .. " [sublevel]" end
    if e.vehicleId then
      tag = tag .. " [on " .. tostring(e.vehicleName or ("Vehicle " .. tostring(e.vehicleId))) .. "]"
    end
  end
  if e.status and e.status ~= "active" then tag = tag .. " [" .. tostring(e.status) .. "]" end
  if e.computerId then tag = tag .. " · PC " .. tostring(e.computerId) end
  return name .. tag
end

local function drawSelect(list, sel, status)
  resetScreen()
  term.setCursorPos(1, 1)
  term.setTextColor(colors.yellow)
  print("=== Assign this GPS (" .. tostring(cfg.role) .. ") ===")
  term.setTextColor(colors.white)
  print("UP / DOWN to highlight, ENTER to confirm.")
  print("")

  if #list == 0 then
    term.setTextColor(colors.red)
    print("Nothing in the registry yet.")
    term.setTextColor(colors.white)
    print("Accept a cannon or vehicle request on the website.")
  else
    local width = term.getSize()
    for i, e in ipairs(list) do
      local line = "  " .. entryLabel(e)
      line = line:sub(1, width)
      line = line .. string.rep(" ", math.max(0, width - #line))
      if i == sel then
        term.setBackgroundColor(colors.blue)
        term.setTextColor(colors.white)
      else
        term.setBackgroundColor(colors.black)
        term.setTextColor(colors.lightGray)
      end
      print(line)
    end
  end

  term.setBackgroundColor(colors.black)
  term.setTextColor(colors.gray)
  print("")
  print(status or "")
  print("vehicle = a Sublevel Vehicle Computer (the whole ship)")
  print("cannon  = one gun running on its own")
  print("auto-refresh every " .. REFRESH_SECONDS .. "s · ESC to cancel")
end

-- Live list (refreshes every REFRESH_SECONDS while the screen is open).
-- Returns the chosen entry, or nil if the user cancelled.
local function selectAssignment()
  local fetched = fetchRegistry()
  local list = fetched or {}
  local status = fetched and "Loaded from the website." or "Could not reach the website."

  local sel = 1
  for i, e in ipairs(list) do
    if assignmentKind() ~= nil and e.kind == assignmentKind()
       and tonumber(e.id) == tonumber(assignmentId()) then
      sel = i
    end
  end

  local timer = os.startTimer(REFRESH_SECONDS)
  drawSelect(list, sel, status)

  while true do
    local event, a = os.pullEvent()
    if event == "timer" and a == timer then
      local fresh = fetchRegistry()
      if fresh then
        list = fresh
        status = "Registry refreshed."
      else
        status = "Website unreachable — keeping the last list."
      end
      if #list == 0 then sel = 1
      elseif sel > #list then sel = #list
      elseif sel < 1 then sel = 1 end
      drawSelect(list, sel, status)
      timer = os.startTimer(REFRESH_SECONDS)
    elseif event == "key" then
      if a == keys.up then
        if sel > 1 then sel = sel - 1 end
        drawSelect(list, sel, status)
      elseif a == keys.down then
        if sel < #list then sel = sel + 1 end
        drawSelect(list, sel, status)
      elseif a == keys.enter then
        if #list > 0 then return list[sel] end
      elseif a == keys.escape then
        return nil
      end
    end
  end
end

-- Opens the list and stores the choice. Returns true when assigned.
local function assignTarget()
  local chosen = selectAssignment()
  if not chosen then return false end

  -- One choice, not a set: assigning to a vehicle clears any cannon
  -- assignment and vice versa, so a beacon can never report for two things.
  cfg.vehicleId, cfg.vehicleName = nil, nil
  cfg.cannonId, cfg.cannonName = nil, nil

  if chosen.kind == "vehicle" then
    cfg.vehicleId   = tonumber(chosen.id)
    cfg.vehicleName = chosen.name
    if not cfg.vehicleName or cfg.vehicleName == "" then
      cfg.vehicleName = "Vehicle " .. tostring(cfg.vehicleId)
    end
  else
    cfg.cannonId   = tonumber(chosen.id)
    cfg.cannonName = chosen.name
    if not cfg.cannonName or cfg.cannonName == "" then
      cfg.cannonName = "Cannon " .. tostring(cfg.cannonId)
    end
  end

  writeConfig()
  resetScreen()
  term.setCursorPos(1, 1)
  term.setTextColor(colors.lime)
  print("Assigned to " .. assignmentLabel() .. ".")
  term.setTextColor(colors.white)
  sleep(1.5)
  resetScreen()
  return true
end

-- ------------------------------------------------------------
-- GPS engine
-- ------------------------------------------------------------
-- os.loadAPI loads gpsnet.lua and exposes it as the global `gpsnet`.
local function loadGpsnet()
  if type(gpsnet) ~= "table" then
    pcall(os.loadAPI, "gpsnet.lua")
  end
  if type(gpsnet) ~= "table" then
    resetScreen()
    term.setCursorPos(1, 1)
    term.setTextColor(colors.red)
    print("GPS Network library missing.")
    term.setTextColor(colors.white)
    print("")
    print("This program needs gpsnet.lua in the same folder.")
    print("Download it as gpsnet.lua, next to this program:")
    print("")
    print("    wget https://raw.githubusercontent.com/starpanda24825/RA/")
    print("         refs/heads/main/luatxtfiles/Cannon%20Programs/")
    print("         GPS%20Network%20V1.txt gpsnet.lua")
    print("")
    print("Press any key to exit.")
    os.pullEvent("key")
    error("gpsnet.lua not installed", 0)
  end
  return gpsnet
end

local gpsnet = loadGpsnet()

-- The beacon drives its own pings (on SYNC ticks), so the engine's automatic
-- schedule is pushed far out of the way.
local engine = gpsnet.new({ interval = 3600, window = 0.5, maxAge = 5 })

-- ------------------------------------------------------------
-- Shared state (written by the modem coroutine, read by the beacon loop)
-- ------------------------------------------------------------
local pendingSync = nil      -- seq of the SYNC tick we still owe an answer to
local syncSeenAt  = nil      -- when that tick arrived
local lastSyncAt  = nil      -- when ANY sync last arrived (for the screen)
local netError    = nil

local stats = {
  sent      = 0,   -- broadcasts made
  missed    = 0,   -- times we wanted a fix and could not get one
  lastSeq   = nil,
  lastSentAt = nil,
}

-- ------------------------------------------------------------
-- Modem coroutine
-- ------------------------------------------------------------
-- One coroutine owns the modem. Every modem_message goes to the GPS engine,
-- and anything on the ship channel that is a SYNC tick is recorded. Nothing
-- blocks on a fix here, so this loop never stops listening.
local function networkPump()
  while true do
    local ok, err = pcall(function()
      engine:attach()
      engine:openChannel(CHANNEL)
      while true do
        local _, side, channel, replyChannel, message, distance = os.pullEvent("modem_message")
        -- Argument order is (side, channel, message, distance). replyChannel
        -- is NOT part of it: the engine's channel is already its own, so it
        -- filters on `channel` alone.
        engine:onMessage(side, channel, message, distance)

        if channel == CHANNEL and type(message) == "string" then
          local parsed, data = pcall(textutils.unserializeJSON, message)
          if parsed and type(data) == "table" and data.type == "ship_sync" then
            local seq = tonumber(data.seq)
            if seq and (pendingSync == nil or seq ~= pendingSync) then
              pendingSync = seq
              syncSeenAt = os.clock()
            end
            lastSyncAt = os.clock()
          end
        end
      end
    end)
    -- A detached or replaced modem lands here; reattach and carry on rather
    -- than taking the whole daemon down with it.
    netError = ok and nil or tostring(err)
    sleep(1)
  end
end

-- ------------------------------------------------------------
-- Fix + broadcast
-- ------------------------------------------------------------
-- Force a fresh ping and wait a moment for it. Returns x, y, z, meta (the
-- meta may describe a slightly older fix if nothing new arrived in time).
local function takeFix(maxWait)
  engine:request()
  local deadline = os.clock() + (maxWait or SYNC_WAIT)
  while os.clock() < deadline do
    local x, y, z, meta = engine:get()
    if x and meta.age <= FIX_FRESH_AGE then return x, y, z, meta end
    engine:update()
    sleep(0.1)
  end
  return engine:get()
end

local function broadcast(x, y, z, meta, seq)
  if not engine.modem then return false end
  local payload = textutils.serializeJSON({
    type      = "ship_gps",
    role      = cfg.role,
    -- Only one of these is ever set. A vehicle names the ship itself; a cannon
    -- names a gun that runs standalone. Receivers accept either.
    vehicleId = cfg.vehicleId,
    cannonId  = cfg.cannonId,
    x         = x,
    y         = y,
    z         = z,
    age       = meta and meta.age or nil,   -- staleness when sent
    seq       = seq,                        -- which SYNC tick this answers
  })
  local ok = pcall(function() engine.modem.transmit(CHANNEL, CHANNEL, payload) end)
  if ok then
    stats.sent = stats.sent + 1
    stats.lastSeq = seq
    stats.lastSentAt = os.clock()
  end
  return ok
end

local function ageText(seconds)
  if not seconds then return "—" end
  if seconds < 1 then return string.format("%.0fms", seconds * 1000) end
  return string.format("%.1fs", seconds)
end

-- ------------------------------------------------------------
-- Beacon loop (screen + broadcast cadence + the C key)
-- ------------------------------------------------------------
local function mainLoop()
  local lastSentAt = -1e9

  while true do
    engine:update()
    engine:openChannel(CHANNEL)      -- reopens itself if the modem is swapped

    -- Answer a SYNC tick immediately: that is what makes the other beacon's
    -- fix and ours describe the same instant.
    local now = os.clock()
    local seq = pendingSync
    local sent = false
    if seq ~= nil and (syncSeenAt == nil or now - syncSeenAt <= SYNC_WAIT + 0.4) then
      local x, y, z, meta = takeFix(SYNC_WAIT)
      if x and broadcast(x, y, z, meta, seq) then
        lastSentAt = os.clock()
        sent = true
      end
    end
    if seq ~= nil then pendingSync = nil end

    -- Otherwise free-run, so a ship still reports if the cannon computer is
    -- off, rebooting, or out of range.
    if not sent and now - lastSentAt >= REPORT_INTERVAL then
      local x, y, z, meta = takeFix(SYNC_WAIT)
      if x then
        if broadcast(x, y, z, meta, nil) then lastSentAt = os.clock() end
      else
        stats.missed = stats.missed + 1
      end
    end

    local x, y, z, meta = engine:get()
    local lines = {}
    lines[#lines + 1] = { "=== Sublevel Ship GPS — " .. tostring(cfg.role) .. " ===", colors.yellow }
    if assignmentName() then
      lines[#lines + 1] = "Assigned to: " .. assignmentName()
        .. " (" .. tostring(assignmentKind()) .. ")"
    else
      lines[#lines + 1] = { "Assigned to: — none — press C to choose", colors.red }
    end
    lines[#lines + 1] = ""

    if x then
      lines[#lines + 1] = { ("Position:  %d, %d, %d"):format(
        math.floor(x + 0.5), math.floor(y + 0.5), math.floor(z + 0.5)), colors.white }
      lines[#lines + 1] = ("Fix:       %d tower(s), %s, layout %d%%, %s old"):format(
        meta.towers, meta.quality, math.floor((meta.geometry or 0) * 100), ageText(meta.age))
      if meta.excluded and #meta.excluded > 0 then
        lines[#lines + 1] = { "Ignored:   tower at " .. table.concat(meta.excluded, ", ") ..
          " does not fit the others", colors.orange }
      end
    else
      lines[#lines + 1] = { "Position:  no fix yet", colors.red }
      lines[#lines + 1] = { "Reason:    " .. tostring(engine.reason or "locating…"), colors.red }
    end

    lines[#lines + 1] = ""
    local since = lastSyncAt and (os.clock() - lastSyncAt) or nil
    if since and since < 15 then
      lines[#lines + 1] = { "Ship controller: in contact (" .. ageText(since) .. " ago) — fixes are "
        .. "sampled together.", colors.lime }
    elseif stats.sent > 0 then
      lines[#lines + 1] = { "Ship controller: not heard from — free-running every "
        .. REPORT_INTERVAL .. "s.", colors.gray }
    else
      lines[#lines + 1] = { "Waiting for a SYNC tick from the ship…", colors.gray }
    end

    lines[#lines + 1] = ("Broadcast: %d sent%s"):format(stats.sent,
      stats.lastSentAt and (" — last " .. ageText(os.clock() - stats.lastSentAt) .. " ago") or "")
    if stats.missed > 0 then
      lines[#lines + 1] = { "Missed:    " .. stats.missed .. " (no fix available)", colors.orange }
    end
    if netError then
      lines[#lines + 1] = { "Network:   " .. netError, colors.red }
    end

    lines[#lines + 1] = ""
    lines[#lines + 1] = { "Channel " .. CHANNEL .. " · press C to change the assignment.", colors.gray }

    drawScreen(lines)

    -- Short tick, and watch for keys so C is answered promptly.
    local timer = os.startTimer(0.25)
    local handled = false
    while not handled do
      local event, a = os.pullEvent()
      if event == "timer" and a == timer then
        handled = true
      elseif event == "key" and a == keys.c then
        assignTarget()
        handled = true
      end
    end
  end
end

-- Main

resetScreen()
readConfig()

if not cfg.role then
  setupRole()
end

-- Require an assignment before the beacon starts reporting.
while not cfg.cannonId and not cfg.vehicleId do
  resetScreen()
  term.setCursorPos(1, 1)
  print("=== Sublevel Ship GPS — assignment ===")
  print("")
  print("This beacon (" .. tostring(cfg.role) .. ") must be assigned to the")
  print("ship it reports the heading for: that ship's Sublevel Vehicle")
  print("Computer, or a cannon that runs standalone.")
  print("")
  print("Press any key to open the registry list…")
  os.pullEvent("key")
  assignTarget()
end

-- First-boot GPS self-check, so a tower problem is obvious immediately rather
-- than showing up as a mysteriously wrong heading later.
do
  resetScreen()
  term.setCursorPos(1, 1)
  term.setTextColor(colors.yellow)
  print("=== GPS self-check (this needs gpsnet.lua) ===")
  term.setTextColor(colors.white)
  local lines, ok = gpsnet.diagnose(engine)
  for _, line in ipairs(lines) do print(line) end
  print("")
  if ok then
    term.setTextColor(colors.lime)
    print("Beacon starting…")
  else
    term.setTextColor(colors.orange)
    print("The beacon will keep retrying in the background.")
  end
  term.setTextColor(colors.white)
  sleep(3)
end

resetScreen()

-- blocks on a fix, so a slow tower never freezes the beacon.
