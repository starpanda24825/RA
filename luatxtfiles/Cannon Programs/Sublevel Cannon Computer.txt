--   - For ship heading: TWO more computers running the "Sublevel Ship

--   3. Wire the cannon's firing mechanism. Two options:
--        a) Peripheral fire (recommended) — cbcperipheral's fire()

--        b) Redstone fire — run a redstone cable from a side of the
--           computer to the firing block and set fireMode=redstone
--           in subcannon_cfg.txt.
--        c) Reload trigger — run a redstone cable from a side of the
--           computer to the reloading mechanism. First boot asks which
--           side this impulse is output on.
--   4. Make sure the cannon is LOADED (shell + powder) before firing.
--
-- RUN
--   1. Put this program on the computer as "startup" (it is a daemon).
--   2. First boot prompts for: cannon length (blocks) and the
--      direction the cannon faces when assembled. Saved to
--      subcannon_cfg.txt. This cannon is always reported as sublevel.
--   3. The computer registers itself with the website as a SUBLEVEL
--      cannon. It shows "Awaiting approval" until an officer accepts
--      it in the Ballistic Calculator → Cannon Registry tab.
--   4. Once accepted it does two jobs at the same time:

--              disassemble → assemble → aim → fire → disassemble
--        b) A concurrent GPS engine keeps a validated fix flowing, and the
--           main loop pushes the freshest coordinates to the website

--
-- DIRECTION FACED WHEN ASSEMBLED ("facing")

--      0   = South (+Z)
--      90  = West
--      180 = North
--      270 = East

-- CALIBRATION

--   offset, tune these in subcannon_cfg.txt:
--     yawOffset  — added to the incoming yaw (degrees)
--     pitchScale — multiplied by the incoming pitch (1 or -1)
-- ============================================================

-- ------------------------------------------------------------

local function trim(s)
  s = tostring(s or "")
  s = s:gsub("^%s+", "")
  s = s:gsub("%s+$", "")
  return s
end

local function randomHex(len)
  local chars = "0123456789abcdef"
  local s = ""
  for i = 1, len do
    local r = math.random(1, #chars)
    s = s .. chars:sub(r, r)
  end
  return s
end

local function readConfig()
  if not fs.exists(CFG_PATH) then return nil end
  local f = fs.open(CFG_PATH, "r")
  if not f then return nil end
  local loaded = {}
  while true do
    local line = f.readLine()
    if line == nil then break end
    local k, v = line:match("^([%w_]+)%s*=%s*(.-)%s*$")
    if k then loaded[k] = v end
  end
  f.close()
  return loaded
end

local function writeConfig()
  local f = fs.open(CFG_PATH, "w")
  f.writeLine("length=" .. tostring(cfg.length))
  f.writeLine("facing=" .. tostring(cfg.facing))
  f.writeLine("reloadType=" .. tostring(cfg.reloadType))
  f.writeLine("reloadTime=" .. tostring(cfg.reloadTime))
  f.writeLine("message=" .. tostring(cfg.message))
  f.writeLine("pollInterval=" .. tostring(cfg.pollInterval))
  f.writeLine("yawOffset=" .. tostring(cfg.yawOffset))
  f.writeLine("pitchScale=" .. tostring(cfg.pitchScale))
  f.writeLine("shipRelativeYaw=" .. tostring(cfg.shipRelativeYaw))
  f.writeLine("fireMode=" .. tostring(cfg.fireMode))
  f.writeLine("redstoneSide=" .. tostring(cfg.redstoneSide))
  f.writeLine("firePulse=" .. tostring(cfg.firePulse))
  f.writeLine("reloadRedstoneSide=" .. tostring(cfg.reloadRedstoneSide))
  f.writeLine("reloadPulse=" .. tostring(cfg.reloadPulse))
  f.close()
end

local function loadConfig()
  local loaded = readConfig()
  if loaded then
    cfg.length       = tonumber(loaded.length) or cfg.length
    cfg.facing       = tonumber(loaded.facing) or cfg.facing
    if loaded.reloadType == "arm" or loaded.reloadType == "autoloader" then
      cfg.reloadType = loaded.reloadType
    end
    cfg.reloadTime   = tonumber(loaded.reloadTime) or cfg.reloadTime
    cfg.message      = loaded.message or cfg.message
    cfg.pollInterval = tonumber(loaded.pollInterval) or cfg.pollInterval
    cfg.yawOffset    = tonumber(loaded.yawOffset) or cfg.yawOffset
    cfg.pitchScale   = tonumber(loaded.pitchScale) or cfg.pitchScale
    if loaded.shipRelativeYaw ~= nil then
      cfg.shipRelativeYaw = (loaded.shipRelativeYaw ~= "false")
    end
    cfg.fireMode     = loaded.fireMode or cfg.fireMode
    cfg.redstoneSide = loaded.redstoneSide or cfg.redstoneSide
    cfg.firePulse    = tonumber(loaded.firePulse) or cfg.firePulse
    cfg.reloadRedstoneSide = loaded.reloadRedstoneSide or cfg.reloadRedstoneSide
    cfg.reloadPulse  = tonumber(loaded.reloadPulse) or cfg.reloadPulse
  end
end

-- ------------------------------------------------------------
-- Computer identity
-- ------------------------------------------------------------
local function resolveComputerId()
  local id = nil
  if os.getComputerID then
    local cid = os.getComputerID()
    if cid and cid ~= 0 then id = tostring(cid) end
  end
  if not id and fs.exists(ID_PATH) then
    local f = fs.open(ID_PATH, "r")
    if f then
      id = trim(f.readAll())
      f.close()
    end
  end
  if not id or id == "" then
    id = randomHex(16)
    local f = fs.open(ID_PATH, "w")
    f.write(id)
    f.close()
  end
  return id
end

local computerId = resolveComputerId()

-- ------------------------------------------------------------
-- Mount peripheral discovery
-- ------------------------------------------------------------
local function findMount()
  if peripheral.find then
    local name, wrapped = peripheral.find(MOUNT_TYPE)
    if wrapped then return wrapped end
  end
  for _, side in ipairs(peripheral.getNames()) do
    if peripheral.getType(side) == MOUNT_TYPE then
      return peripheral.wrap(side)
    end
  end
  return nil
end

-- ------------------------------------------------------------
-- GPS location
-- ------------------------------------------------------------
-- Positions come from the gpsnet engine — the "GPS Network" library, which
-- must be installed next to this program as gpsnet.lua. It replaces the
-- stock gps.locate() with something far better behaved:
--
--   * It solves the WHOLE tower set at once (least squares) instead of taking
--     three towers and hoping, so a fix normally lands in four replies rather
--     than running out a two-second timeout.
--   * It checks the answer against its own residuals and throws out any tower
--     that does not fit, so a tower that has been moved, mis-keyed or is
--     answering from somewhere else is DETECTED instead of quietly bending
--     the position — and with it the aim.
--   * Replies come back on a private channel, so this computer hears its own
--     towers and nothing else. Stock gps.locate() asks for replies on the
--     shared channel, which means every receiver in range is sent every
--     reply; that is fine for a handful of computers and hopeless for a
--     network.
--   * The newest fix is cached, so nothing ever blocks waiting for one. The
--     main loop reads a position and its age and carries on.
--
-- A fix needs FOUR OR MORE towers, at least one of them clearly off the plane
-- of the others (a different height). Three towers, or a set of towers all at
-- one height, cannot determine a position at all; the engine says so rather
-- than guessing. See the GPS Network file for the details.

local function loadGpsnet()
  if type(gpsnet) ~= "table" then
    pcall(os.loadAPI, "gpsnet.lua")
  end
  if type(gpsnet) ~= "table" then
    term.setBackgroundColor(colors.black)
    term.clear()
    term.setCursorPos(1, 1)
    term.setTextColor(colors.red)
    print("GPS Network library missing (gpsnet.lua).")
    term.setTextColor(colors.white)
    print("")
    print("Download the \"GPS Network\" file next to this program as")
    print("gpsnet.lua:")
    print("")
    print("    wget https://raw.githubusercontent.com/starpanda24825/RA/")
    print("         refs/heads/main/luatxtfiles/Cannon%20Programs/")
    print("         GPS%20Network%20V1.txt gpsnet.lua")
    print("")
    print("Press any key to exit.")
    os.pullEvent("key")
    error("gpsnet.lua not installed", 0)
  end
  return gpsnet
end

local gpsnet = loadGpsnet()
local engine = gpsnet.new({ interval = GPS_INTERVAL, window = GPS_WINDOW, maxAge = 4 })

-- Concurrent GPS scheduler. The modem itself is pumped by networkPump()
-- below; this only drives the ping schedule and expires stale fixes, and it
-- never throws, so one bad attempt cannot take the daemon down.
local function gpsScheduler()
  while true do
    local ok, err = pcall(function() engine:update() end)
    if not ok then engine.reason = "GPS error: " .. tostring(err) end
    sleep(0.25)
  end
end

-- First-boot GPS self-check. It runs with the modem coroutine already live
-- (see the bottom of this file) so it can genuinely obtain a fix, and it
-- prints the specific reason a fix is or is not available.
local function gpsSelfCheck()
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setCursorPos(1, 1)
  term.setTextColor(colors.yellow)
  print("=== GPS self-check ===")
  term.setTextColor(colors.white)
  local lines, ok = gpsnet.diagnose(engine)
  for _, line in ipairs(lines) do print(line) end
  return ok
end

-- ------------------------------------------------------------
-- Ship heading (two-point GPS)
-- ------------------------------------------------------------
-- Two dedicated computers run the "Sublevel Ship GPS" program: one at the
-- FRONT (bow) and one at the BACK (stern). Both broadcast their position on
-- the shared channel; this cannon computer listens to both and derives the
-- ship's heading as the direction from back to front, in the same yaw
-- convention as the website (0 = south, 90 = west, 180 = north, 270 = east).
local SHIP_GPS_CHANNEL = 4120   -- must match the Ship GPS program

-- Channel this cannon's Sublevel Vehicle Computer talks to it on, once the
-- website has assigned this cannon to a vehicle. MUST match VEHICLE_CHANNEL in
-- the Sublevel Vehicle Computer program.
local VEHICLE_CHANNEL = 4121

-- MANAGED MODE
--   When the website reports that a Sublevel Vehicle Computer owns this cannon,
--   this computer stops doing aiming maths. The vehicle holds the ship's
--   heading — ONE value shared by every gun on the ship, rather than each
--   computer deriving its own — and it hands this cannon its final mount aim.
--   All this computer does then is apply the values it is given.
local managedVehicle = nil    -- { id, name, computerId }, from the poll response
local pendingAim     = nil    -- the next aim the vehicle has assigned
local vehicleShipYaw = nil    -- the heading the vehicle is using
local lastVehicleAt  = nil    -- when the vehicle last spoke to us

-- The registry id of THIS cannon on the website, learned from the poll
-- response (see mainLoop). Every Ship GPS beacon names the cannon it is
-- assigned to, and we only accept beacons that name us. Without this, two
-- ships in wireless range would overwrite each other's front/back fixes and
-- the heading (and therefore the aim) would belong to the wrong vehicle.
local myCannonId = nil

-- The two beacons' most recent accepted fixes. Each carries how old its fix
-- was when the beacon sent it (age) and which SYNC tick it answered (seq), so
-- the heading can be judged as well as calculated.
local BEACON_MAX_AGE = 8     -- seconds before a beacon's fix is too old to use
local SYNC_TOLERANCE = 0.75  -- seconds of mismatch worth warning about

local front = nil   -- { x, y, z, age, seq, at }
local back  = nil

local gpsNetError = nil

-- math.atan2 exists in CC:Tweaked (Lua 5.1 semantics), but fall back to the
-- two-argument math.atan so heading math can't silently break on an upgrade.
local atan2 = math.atan2
if not atan2 and math.atan then
  atan2 = function(y, x) return math.atan(y, x) end
end

-- Ship heading, as the direction from the stern fix to the bow fix, in the
-- website's convention (0 = south, 90 = west, 180 = north, 270 = east).
--
-- Also returns how far apart in time the two fixes were taken. The heading is
-- only exact if both points describe the SAME instant: if the ship turned
-- between them, the bow-to-stern line is rotated by roughly half the turn
-- rate times that gap. A ship turning at 10°/s with fixes a second apart is
-- about 5° out, which matters a great deal at long range. This computer
-- broadcasts a SYNC tick so both beacons sample together; the returned skew
-- is the number that says whether they did.
local function computeShipYaw()
  if not front or not back or not atan2 then return nil, nil end
  local dx = front.x - back.x
  local dz = front.z - back.z
  if math.abs(dx) < 0.001 and math.abs(dz) < 0.001 then return nil, nil end

  local now = os.clock()
  if (now - front.at) > BEACON_MAX_AGE or (now - back.at) > BEACON_MAX_AGE then
    return nil, nil
  end

  local deg = math.deg(atan2(-dx, dz))
  local yaw = ((deg % 360) + 360) % 360

  local skew
  if front.seq and back.seq and front.seq == back.seq then
    -- Both answered the same tick, so the gap is just the difference in how
    -- long each beacon's round trip took — a fraction of a tick.
    skew = math.abs((front.age or 0) - (back.age or 0))
  else
    -- Different ticks, or an older beacon that sends no seq: fall back to the
    -- gap between when the two broadcasts arrived here.
    skew = math.abs(front.at - back.at)
  end
  return yaw, skew
end

-- Concurrent modem pump. ONE coroutine owns the modem and deals with
-- everything that arrives on it:
--
--   * GPS replies are handed straight to the engine;
--   * broadcasts from the ship's two GPS computers are recorded, but only
--     when they are addressed to THIS cannon (data.cannonId == myCannonId).
--     Without that check two ships within wireless range would overwrite each
--     other's front/back fixes, and the aim would belong to the wrong
--     vehicle;
--   * SYNC ticks from the website are ignored here (they are ours).
--
-- The body is pcall'd so a detached or replaced modem restarts the pump
-- rather than taking the whole daemon down (parallel.waitForAll rethrows).
local function networkPump()
  while true do
    local ok, err = pcall(function()
      engine:attach()
      engine:openChannel(SHIP_GPS_CHANNEL)
      engine:openChannel(VEHICLE_CHANNEL)
      while true do
        local _, side, channel, replyChannel, message, distance = os.pullEvent("modem_message")
        -- Argument order is (side, channel, message, distance). replyChannel
        -- is NOT part of it: the engine's channel is already its own, so it
        -- filters on `channel` alone.
        engine:onMessage(side, channel, message, distance)

        if channel == SHIP_GPS_CHANNEL and type(message) == "string" then
          local parsed, data = pcall(textutils.unserializeJSON, message)
          if parsed and type(data) == "table" and data.type == "ship_gps"
             and data.x and data.z and myCannonId ~= nil
             and tonumber(data.cannonId) == tonumber(myCannonId) then
            local fix = {
              x = tonumber(data.x), y = tonumber(data.y), z = tonumber(data.z),
              age = tonumber(data.age), seq = tonumber(data.seq),
              at = os.clock(),
            }
            if data.role == "back" then back = fix else front = fix end
          end
        elseif channel == VEHICLE_CHANNEL and type(message) == "string" then
          -- An aim assigned by our vehicle. Addressing is by registry id, so a
          -- second vehicle in wireless range can never aim this cannon.
          local parsed, data = pcall(textutils.unserializeJSON, message)
          if parsed and type(data) == "table" and data.type == "vehicle_aim"
             and myCannonId ~= nil and tonumber(data.cannonId) == tonumber(myCannonId) then
            pendingAim = {
              sequence = tonumber(data.sequence) or 0,
              yaw = tonumber(data.yaw) or 0,
              pitch = tonumber(data.pitch) or 0,
              fire = data.fire ~= false,
              fromVehicle = true,     -- already ship-relative: do NOT correct again
            }
            vehicleShipYaw = tonumber(data.shipYaw)
            lastVehicleAt = os.clock()
          end
        end
      end
    end)
    gpsNetError = ok and nil or tostring(err)
    sleep(1)
  end
end

-- Tell both beacons to take a reading at the same instant. Each answers a
-- SYNC tick immediately, so their positions come from one moment and the
-- heading between them is a rigid-body measurement rather than two samples of
-- a turning ship. One small message per SHIP_SYNC_INTERVAL is all it costs.
local syncSeq = 0
local function shipSyncBroadcaster()
  while true do
    -- A managed cannon has handed this job to its vehicle: the vehicle sends
    -- the one SYNC tick that keeps the whole ship's beacons sampling together,
    -- and two tickers on the same channel would just be noise.
    if managedVehicle == nil and engine:openChannel(SHIP_GPS_CHANNEL) and engine.modem then
      syncSeq = syncSeq + 1
      local payload = textutils.serializeJSON({ type = "ship_sync", seq = syncSeq })
      pcall(function()
        engine.modem.transmit(SHIP_GPS_CHANNEL, SHIP_GPS_CHANNEL, payload)
      end)
    end
    sleep(SHIP_SYNC_INTERVAL)
  end
end

-- Report this cannon's live position and aim state to our vehicle. The vehicle
-- forwards all of it to the website on our behalf and aims us from this
-- cannon's own position, so while it is in charge this computer neither polls
-- for commands nor works out an aim of its own.
local function vehicleReport(hasMount, x, y, z, gpsOk, yaw, pitch, sequence)
  if not engine:openChannel(VEHICLE_CHANNEL) or not engine.modem then return false end
  if myCannonId == nil then return false end
  local payload = textutils.serializeJSON({
    type = "cannon_report",
    cannonId = myCannonId,
    computerId = computerId,
    -- A cannon with no fix yet reports no position at all, so the registry
    -- keeps its last known coordinates instead of jumping to 0,0,0.
    x = gpsOk and x or nil, y = gpsOk and y or nil, z = gpsOk and z or nil,
    gpsOk = gpsOk == true,
    yaw = yaw, pitch = pitch,
    sequence = sequence,
    mount = hasMount == true,
  })
  return pcall(function() engine.modem.transmit(VEHICLE_CHANNEL, VEHICLE_CHANNEL, payload) end)
end

-- A compact health summary of the GPS network, piggy-backed on the poll we
-- already make every second. That way the website can show the state of the
-- tower network without a single extra request, however big the network gets.
-- The tower list itself is only sent when it actually changes.
local towerSignature = nil
local towersSentAt = 0

local function gpsReport(meta, skew)
  local report = { reason = tostring(engine.reason or "") }
  if meta then
    report.ok       = not meta.stale
    report.quality  = meta.quality
    report.geometry = meta.geometry
    report.residual = meta.rms
    report.towers   = meta.towers
    report.seen     = meta.seen
    report.age      = meta.age
    report.stale    = meta.stale
    report.shipSkew = skew
    if meta.excluded and #meta.excluded > 0 then report.excluded = meta.excluded end
    report.reason = (engine.reason ~= nil) and tostring(engine.reason) or nil
  else
    report.ok = false
  end

  local list = engine:towerList()
  local parts = {}
  for i = 1, #list do
    parts[i] = list[i].x .. "," .. list[i].y .. "," .. list[i].z
  end
  table.sort(parts)
  local signature = table.concat(parts, ";")
  if signature ~= towerSignature or (os.clock() - towersSentAt) > 60 then
    towerSignature = signature
    towersSentAt = os.clock()
    report.towerList = list
    report.towerSignature = signature
  end
  return report
end

-- ------------------------------------------------------------
-- HTTP bridge
-- ------------------------------------------------------------
local function httpPost(path, body)
  local url = BASE_URL .. path
  local headers = { ["Content-Type"] = "application/json" }
  local bodyStr = textutils.serializeJSON(body)
  local ok, result = pcall(function()
    local req = http.post(url, bodyStr, headers)
    if req then
      local raw = req.readAll()
      req.close()
      return raw
    end
    return nil
  end)
  if not ok or result == nil then return nil end
  local ok2, decoded = pcall(textutils.unserializeJSON, result)
  if ok2 and decoded then return decoded end
  return nil
end

-- ------------------------------------------------------------
-- Firing sequence (same as static cannons):
--   disassemble → assemble → aim → fire → disassemble
-- ------------------------------------------------------------
local function normDeg(d)
  d = d % 360
  if d > 180 then d = d - 360 end
  return d
end

local function waitAimed(mount, aimYaw, aimPitch)
  local start = os.clock()
  while os.clock() - start < AIM_TIMEOUT do
    local y, p
    pcall(function() y = mount.getYaw() end)
    pcall(function() p = mount.getPitch() end)
    if y ~= nil and p ~= nil then
      local dy = math.abs(normDeg(y - aimYaw))
      local dp = math.abs(p - aimPitch)
      if dy <= AIM_TOLERANCE and dp <= AIM_TOLERANCE then
        return true
      end
    end
    sleep(0.1)
  end
  return false
end

local function doFire(mount)
  if cfg.fireMode == "redstone" and cfg.redstoneSide ~= "" then
    redstone.setOutput(cfg.redstoneSide, true)
    sleep(cfg.firePulse)
    redstone.setOutput(cfg.redstoneSide, false)
    return true
  end
  return pcall(function() mount.fire() end)
end

local function triggerReload()
  if cfg.reloadRedstoneSide and cfg.reloadRedstoneSide ~= "" then
    redstone.setOutput(cfg.reloadRedstoneSide, true)
    sleep(cfg.reloadPulse)
    redstone.setOutput(cfg.reloadRedstoneSide, false)
  end
end

local function runCommand(mount, cmd)
  print("Command #" .. tostring(cmd.sequence) .. ": aiming…")

  -- 1. Disassemble. An auto-loader reloads here, between disassemble and
  -- assemble.
  pcall(function() mount.disassemble() end)
  if cfg.reloadType == "autoloader" then
    triggerReload()
    sleep(cfg.reloadTime)
  end

  -- 2. Assemble
  pcall(function() mount.assemble() end)
  sleep(SEQ_ASSEMBLE)

  -- A mechanical arm reloads after the cannon is assembled.
  if cfg.reloadType == "arm" then
    triggerReload()
    sleep(cfg.reloadTime)
  end

  -- 3. Delay before aiming
  sleep(SEQ_BEFORE_AIM)

  -- 4. Aim. The website sends an absolute world yaw. On a sublevel ship the
  -- mount's yaw is measured in the ship's rotating frame, so the barrel's
  -- world heading is (shipYaw + mountYaw). To hit an absolute world heading we
  -- must therefore SUBTRACT the ship's heading. (Adding it mirrored the shot
  -- by 2 x shipYaw — an almost 180° error whenever the ship faced E or W,
  -- which is exactly the "aimed the opposite way" symptom.)
  --
  -- Recompute the heading here, at aim time, rather than reusing the value
  -- sampled before the poll: disassemble/assemble/reload can take ~10s and the
  -- ship may have turned since.
  -- A command assigned by a vehicle already has the ship's heading taken out
  -- of it (the vehicle owns the heading, and it is the same heading for every
  -- gun on the ship), so correcting it again here would double the error.
  local aimYaw = (cmd.yaw or 0) + cfg.yawOffset
  if not cmd.fromVehicle then
    local shipYaw = computeShipYaw()
    if cfg.shipRelativeYaw and shipYaw == nil then
      print("Warning: ship heading unknown — aiming in world space.")
    end
    if cfg.shipRelativeYaw and shipYaw then
      aimYaw = aimYaw - shipYaw
    end
  end
  aimYaw = ((aimYaw % 360) + 360) % 360
  local aimPitch = (cmd.pitch or 0) * cfg.pitchScale
  pcall(function() mount.setYaw(aimYaw) end)
  pcall(function() mount.setPitch(aimPitch) end)
  local settled = waitAimed(mount, aimYaw, aimPitch)
  if not settled then
    print("Warning: aim did not settle — firing anyway")
  end

  -- 5. Delay before firing
  sleep(SEQ_BEFORE_FIRE)

  -- 6. Fire
  local fired = doFire(mount)
  if fired then
    print("Fired! (yaw " .. string.format("%.1f", aimYaw) .. ", pitch " .. string.format("%.1f", aimPitch) .. ")")
  else
    print("Fire failed — is the cannon loaded and assembled?")
  end
  sleep(SEQ_AFTER_FIRE)

  -- 7. Disassemble
  pcall(function() mount.disassemble() end)
end

-- ------------------------------------------------------------
-- Status screen
-- ------------------------------------------------------------
local function drawStatus(lines)
  term.setBackgroundColor(colors.black)
  term.clear()
  term.setCursorPos(1, 1)
  term.setTextColor(colors.white)
  local width = term.getSize()
  for _, line in ipairs(lines) do
    -- A line may be a plain string, or { text, colour } when it is worth
    -- drawing attention to.
    local text, colour = line, colors.white
    if type(line) == "table" then
      text, colour = line[1] or "", line[2] or colors.white
    end
    if #text > width then text = text:sub(1, width) end
    term.setTextColor(colour)
    print(text)
  end
  term.setTextColor(colors.white)
end

-- ------------------------------------------------------------
-- Main
-- ------------------------------------------------------------
loadConfig()
if not readConfig() then
  setup()
  loadConfig()
end

local lastSequence = 0

local function mainLoop()
  while true do
    local lines = {}
    local mount = findMount()

    local curYaw, curPitch = 0, 0
    if mount then
      pcall(function() curYaw = mount.getYaw() or 0 end)
      pcall(function() curPitch = mount.getPitch() or 0 end)
    end

    -- Read the newest validated fix from the GPS engine. If we have never had
    -- one, x/y/z stay nil so the website keeps its last known position instead
    -- of snapping the cannon's dot to 0,0,0.
    local gpsX, gpsY, gpsZ, gpsMeta = engine:get()
    local gpsOk = gpsX ~= nil

    -- Managed by a vehicle? Then the heading is the VEHICLE's, not one derived
    -- here: one ship, one heading, every gun. If the vehicle has gone quiet we
    -- show no heading rather than silently falling back to our own, because a
    -- different heading per cannon is exactly what the vehicle exists to stop.
    local managed = managedVehicle ~= nil
    local shipYaw, shipSkew = nil, nil
    if managed then
      if vehicleShipYaw and lastVehicleAt and (os.clock() - lastVehicleAt) <= BEACON_MAX_AGE then
        shipYaw = vehicleShipYaw
      end
    else
      shipYaw, shipSkew = computeShipYaw()
    end

    if managed then
      vehicleReport(mount ~= nil, gpsX, gpsY, gpsZ, gpsOk, curYaw, curPitch, lastSequence)
    end

    local body = {
      computerId = computerId,
      x = gpsX, y = gpsY, z = gpsZ,
      gpsOk = gpsOk,
      shipYaw = shipYaw,
      length = cfg.length,
      facing = cfg.facing,
      sublevel = true,
      message = cfg.message,
      yaw = curYaw,
      pitch = curPitch,
      sequence = lastSequence,
      gps = gpsReport(gpsMeta, shipSkew),
    }

    local resp = httpPost(POLL_PATH, body)

    -- Learn (or refresh) this cannon's registry id from the server so the ship
    -- GPS listener knows which beacons to trust.
    if resp and resp.success and resp.response and resp.response.id then
      myCannonId = tonumber(resp.response.id) or myCannonId
    end

    -- Does a vehicle own this cannon? Only an ACCEPTED vehicle is ever reported
    -- here (the server withholds a pending one), and a cannon that is released
    -- goes straight back to running its own command loop.
    if resp and resp.success and resp.response then
      managedVehicle = resp.response.vehicle or nil
    end

    table.insert(lines, "Sublevel Cannon Computer")
    table.insert(lines, "Computer ID: " .. computerId)
    if gpsX ~= nil then
      table.insert(lines, ("GPS: X %d  Y %d  Z %d  (%d towers, %s, layout %d%%, %.1fs old)"):format(
        math.floor(gpsX + 0.5), math.floor(gpsY + 0.5), math.floor(gpsZ + 0.5),
        gpsMeta.towers, gpsMeta.quality, math.floor((gpsMeta.geometry or 0) * 100), gpsMeta.age))
    else
      table.insert(lines, "GPS: no fix yet")
      table.insert(lines, "GPS issue: " .. tostring(engine.reason or "locating…"))
    end
    if gpsMeta and gpsMeta.excluded and #gpsMeta.excluded > 0 then
      table.insert(lines, "GPS: ignoring tower at " .. table.concat(gpsMeta.excluded, ", "))
    end
    if gpsNetError then
      table.insert(lines, "Network: " .. gpsNetError)
    end
    if shipYaw then
      local skewNote = ""
      if shipSkew then
        if shipSkew <= SYNC_TOLERANCE then
          skewNote = string.format(" (beacons synced %.2fs)", shipSkew)
        else
          skewNote = string.format(" (WARNING: beacons %.2fs apart)", shipSkew)
        end
      end
      table.insert(lines, "Ship heading: " .. string.format("%.1f", shipYaw) .. "°" .. skewNote)
    elseif managed then
      table.insert(lines, "Ship heading: waiting for the vehicle's heading")
    elseif myCannonId == nil then
      table.insert(lines, "Ship heading: waiting for registry id…")
    else
      table.insert(lines, "Ship heading: waiting for assigned front/back GPS…")
    end

    if managedVehicle then
      table.insert(lines, { "Vehicle: " .. tostring(managedVehicle.name or ("Vehicle " .. tostring(managedVehicle.id)))
        .. " — heading and commands routed through it", colors.lime })
    end

    if not mount then
      table.insert(lines, "Mount: not found")
      table.insert(lines, "Place this computer touching the cannon mount.")
    end

    if resp == nil then
      table.insert(lines, "Status: cannot reach website")
      table.insert(lines, "Check " .. BASE_URL)
    elseif not resp.success then
      table.insert(lines, "Status: server error")
      table.insert(lines, tostring(resp.response))
    elseif resp.response.status == "pending" then
      table.insert(lines, "Status: AWAITING APPROVAL")
      table.insert(lines, "An officer must accept this cannon")
      table.insert(lines, "from the website's Cannon Registry tab.")
    elseif resp.response.status == "active" then
      -- A managed cannon is given its aim by the vehicle, never by the server:
      -- the vehicle is the one place the ship-relative correction can be made
      -- with a single shared heading, and it acks the command itself. The
      -- server withholds the command from a managed cannon as well, so the two
      -- together leave exactly one route to firing. Deliberately NOT falling
      -- back to resp.response.command here: an uncorrected world yaw fired as
      -- if it were a mount yaw is precisely the "aimed the opposite way" bug.
      local cmd = pendingAim
      if not managed then cmd = resp.response.command end
      if cmd and cmd.sequence ~= lastSequence then
        if mount then
          lastSequence = cmd.sequence   -- ack intent immediately (at-most-once fire)
          if managed then pendingAim = nil end
          local ok, err = pcall(runCommand, mount, cmd)
          if not ok then
            print("Command #" .. tostring(cmd.sequence) .. " failed: " .. tostring(err))
          end
        else
          table.insert(lines, "Fire command queued — mount not found")
        end
      else
        table.insert(lines, "Status: READY")
        table.insert(lines, "Name: " .. tostring(resp.response.name))
        table.insert(lines, "Aiming: yaw " .. string.format("%.1f", curYaw) .. " / pitch " .. string.format("%.1f", curPitch))
      end
    else
      table.insert(lines, "Status: " .. tostring(resp.response.status))
    end

    drawStatus(lines)

    -- While a vehicle is in charge it reports our position to the website on
    -- our behalf, so this poll is only a status check and does not need to run
    -- every second. Slowing it down while managed means the whole ship costs
    -- the server LESS than the same cannons running standalone would.
    sleep(managed and MANAGED_POLL_INTERVAL or cfg.pollInterval)
  end
end

-- Four coroutines: one runs the GPS engine's schedule, one owns the modem
-- and everything arriving on it, one tells the ship's beacons when to sample,
-- and one polls the website and fires. Every one of them yields rather than
-- blocking, so a slow tower or a slow website can never stall firing.
--
-- The self-check runs first, with the modem coroutine already live, so it can
-- genuinely obtain a fix instead of just timing out.
local function bootSequence()
  local ok = gpsSelfCheck()
  if not ok then
    print("")
    print("The daemon will keep retrying in the background.")
  end
  sleep(3)
  mainLoop()
end

parallel.waitForAll(networkPump, gpsScheduler, shipSyncBroadcaster, bootSequence)
