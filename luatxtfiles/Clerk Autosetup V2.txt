-- Clerk Autosetup
-- Simplified version of ATM Autosetup for the Store Clerk system.
os.pullEvent = os.pullEventRaw -- Lock the shell

local barrels = 0
local modems = 0
local computers = 0
local enderModems = 0

function checklistItem(text, currentAmount, amountNeeded)
    local check = "[ ] "
    if (currentAmount >= amountNeeded) then
        check = "[x] "
    end
    print(check..text..": "..currentAmount.."/"..amountNeeded)
    return currentAmount >= amountNeeded
end

function selectItem(itemName)
    while (true) do
        for i=1, 16 do
            local item = turtle.getItemDetail(i)
            if (item ~= nil) then
                if (item.name == itemName) then
                    turtle.select(i)
                    return true
                end
            end
        end
        print("Missing item: "..itemName)
        os.pullEvent("turtle_inventory")
    end
end

function place() while (not turtle.place()) do sleep(1) end end
function back() while (not turtle.back()) do sleep(1) end end
function forward() while (not turtle.forward()) do sleep(1) end end
function up() while (not turtle.up()) do sleep(1) end end
function down() while (not turtle.down()) do sleep(1) end end

local function main()
term.clear()
term.setCursorPos(1,1)
print("=== Clerk Auto setup ===")
print("")
print("Hello! I am a Clerk Turtle. I handle item movement for the store. I'm compatible with Create 6 systems.")
print("")
print("I will build the clerk terminal for you. I need a barrel in front. Rejected items will be dropped to my right.")
print("")
print("Press any key to continue...")
os.pullEvent("key")

local requiredFuel = 15
while (turtle.getFuelLevel() < requiredFuel) do
    term.clear()
    term.setCursorPos(1,1)
    print("First I'm going to need a bit of fuel.")
    print("")
    print("Fuel level: "..turtle.getFuelLevel().."/"..requiredFuel)
    print("Press any key to consume fuel...")
    os.pullEvent("key")
    print("")
    print("Refuelling...")
    for i=1, 16 do
        if (turtle.getFuelLevel() < requiredFuel) then
            while (turtle.getItemCount(i) > 0) do
                turtle.select(i)
                if (turtle.refuel(1)) then
                    if (turtle.getFuelLevel() >= requiredFuel) then
                        break
                    end
                end
            end
        end
    end
    sleep(0.1)
    if (turtle.getFuelLevel() < requiredFuel) then
        print("I still don't have enough fuel...")
        sleep(2)
    else
        print("All fueled up!")
        sleep(1)
    end
end

while (true) do
    barrels = 0
    modems = 0
    computers = 0
    enderModems = 0
    for i=1, 16 do
        local item = turtle.getItemDetail(i)
        if (item ~= nil) then
            if (item.name == "minecraft:barrel") then barrels = barrels + item.count
            elseif (item.name == "computercraft:wireless_modem_normal") then modems = modems + item.count
            elseif (item.name == "computercraft:wireless_modem_advanced") then enderModems = enderModems + item.count
            elseif (item.name == "computercraft:computer_advanced") then computers = computers + item.count
            end
        end
    end

    local leftEquipment = peripheral.getType("left")
    local rightEquipment = peripheral.getType("right")
    if (leftEquipment == "modem" or rightEquipment == "modem") then
        modems = modems + 1
    end

    term.clear()
    term.setCursorPos(1,1)

    print("What I need:")
    print("")
    local ready = true
    if (not checklistItem("A barrel", barrels, 1)) then ready = false end
    if (not checklistItem("A regular modem", modems, 1)) then ready = false end
    if (not checklistItem("An ender modem", enderModems, 1)) then ready = false end
    if (not checklistItem("An advanced computer", computers, 1)) then ready = false end

    if (ready) then
        print("")
        print("All ready! Press any key to continue...")
        os.pullEvent("key")
        break
    end

    while true do
        local event = os.pullEvent()
        if event == "turtle_inventory" or event == "peripheral" or event == "peripheral_detach" then
            break
        end
    end
end

while (not fs.exists("disk")) do
    term.clear()
    term.setCursorPos(1,1)
    print("Please insert the installer disk into the disk drive. I need it to install the Clerk Computer program.")
    print("Insert installer disk to continue...")
    os.pullEvent("disk")
end

-- Building sequence
term.clear()
term.setCursorPos(1,1)
print("Building Clerk Terminal...")

-- Create install flag for disk
local f = fs.open("disk/installClerkPlease.txt", "w")
f.writeLine("Clerk Setup")
f.close()

back()
selectItem("computercraft:computer_advanced")
place()
sleep(1)
peripheral.call("front", "turnOn")
sleep(1)
up()
up()
forward()
selectItem("computercraft:wireless_modem_advanced")
turtle.placeDown()
sleep(0.5)
forward()
turtle.turnRight()
forward()
down()
down()
turtle.turnRight()
selectItem("minecraft:barrel")
place()

-- 8. Equip normal modem
local leftEquipment = peripheral.getType("left")
local rightEquipment = peripheral.getType("right")
local hasModem = leftEquipment == "modem" or rightEquipment == "modem"
if (not hasModem) then
    selectItem("computercraft:wireless_modem_normal")
    turtle.equipLeft()
end

-- Turtle is now behind the barrel

-- Installation
settings.set("shell.allow_disk_startup", false)
settings.save()
print("Installing Clerk Turtle software...")
fs.delete("bankapi.lua")
shell.run("wget", "https://raw.githubusercontent.com/starpanda24825/RA/refs/heads/main/luatxtfiles/Bank%20API%20V2.txt", "bankapi.lua")
fs.delete("startup.lua")
shell.run("wget", "https://raw.githubusercontent.com/starpanda24825/RA/refs/heads/main/luatxtfiles/Clerk%20Turtle%20V2.txt", "startup.lua")
if fs.exists("disk/Clerk Turtle V2.txt") then
    fs.copy("disk/Clerk Turtle V2.txt", "startup.lua")
end

print("All done!")
sleep(1)
os.reboot()
end

while true do
    local ok, err = pcall(main)
    if not ok then
        term.setBackgroundColor(colors.black)
        term.setTextColor(colors.red)
        term.clear()
        term.setCursorPos(1,1)
        if err == "Terminated" then
            print("Termination prohibited.")
        else
            print("Setup Error: " .. tostring(err))
        end
        print("Retrying in 5 seconds...")
        sleep(5)
    end
end
