-- ============================================================
-- Regnum Aeternum — Exchange Autosetup V1
-- Self-installing script for exchange floor computers.
-- Downloads the terminal or ticker from GitHub, prompts for
-- config (worker domain, token, monitor side), and saves it.
-- ============================================================

local RAW_BASE = "https://raw.githubusercontent.com/starpanda24825/RA/refs/heads/main/luatxtfiles/"

print("=== Fiducia Exchange — Autosetup ===")
print("")
print("1) Exchange Terminal (interactive trading)")
print("2) Exchange Ticker (wall-mounted display)")
print("")
write("Choose [1/2]: ")
local choice = (read() or ""):gsub("%s+$", "")

local filename, description, url
if choice == "2" then
  filename = "ticker"
  description = "Exchange Ticker"
  url = RAW_BASE .. "Exchange%20Ticker%20V1.txt"
else
  filename = "exchange"
  description = "Exchange Terminal"
  url = RAW_BASE .. "Exchange%20Terminal%20V1.txt"
end

-- Worker domain (the API_BASE host the scripts talk to)
print("")
print("Worker domain (no https://, no trailing slash).")
print("Example: regnumaeternum.starpanda24825.workers.dev")
write("> ")
local domain = (read() or ""):gsub("^%s+", ""):gsub("%s+$", "")
if domain == "" then
  domain = "regnumaeternum.starpanda24825.workers.dev"
  print("Using default: " .. domain)
end

print("")
print("Installing " .. description .. "...")

-- Download the script via wget (GitHub raw link)
local outFile = filename .. ".lua"
fs.delete(outFile)
shell.run("wget", url, outFile)

if not fs.exists(outFile) then
  print("Download failed. Check that HTTP is enabled and the URL is reachable.")
  return
end

local content = fs.open(outFile, "r").readAll()
content = content:gsub("YOUR_WORKER_DOMAIN", domain)

-- Token (terminal only)
if choice ~= "2" then
  print("")
  print("Enter your CC token (from the admin panel):")
  write("> ")
  local token = (read() or ""):gsub("^%s+", ""):gsub("%s+$", "")
  if token ~= "" then
    local tf = fs.open(".fdxtoken", "w")
    tf.write(token)
    tf.close()
    print("Token saved to .fdxtoken")
  end
end

-- Monitor side (ticker only)
if choice == "2" then
  print("")
  print("Monitor side (left/right/top/bottom/back/front):")
  write("> ")
  local side = (read() or ""):gsub("^%s+", ""):gsub("%s+$", "")
  if side ~= "" then
    content = content:gsub('MONITOR_SIDE = "right"', 'MONITOR_SIDE = "' .. side .. '"')
    print("Monitor side set to: " .. side)
  end
end

-- Write the patched script back to disk
local f = fs.open(outFile, "w")
f.write(content)
f.close()

-- Create startup script
local sf = fs.open("startup.lua", "w")
sf.write('shell.run("' .. outFile .. '")')
sf.close()

print("")
print("========================================")
print("  Setup complete!")
print("  Run: " .. outFile)
print("  Or reboot to auto-start.")
print("========================================")
